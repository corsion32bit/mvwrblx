local function Connect(signal, callback)
    local connection = signal:Connect(callback)
    table.insert(Connections, connection)
    return connection
end

if _G.FortifiedHubCleanup then
    pcall(_G.FortifiedHubCleanup)
end

_G.FortifiedHubCleanup = function()
    for _, connection in ipairs(Connections) do
        if connection and connection.Connected then
            connection:Disconnect()
        end
    end
    table.clear(Connections)
    if CSHandsDestroy then
        pcall(CSHandsDestroy)
    end
    if RestoreTextureOptimization then
        RestoreTextureOptimization()
    end
    if RestoreStrafeOptimizerSensitivity then
        pcall(RestoreStrafeOptimizerSensitivity)
    end
    if StopKeyboardSpammer then
        pcall(function() StopKeyboardSpammer("HUB RELOADED") end)
    end
    if StopWallGlide then
        pcall(StopWallGlide)
    end
    if StopWallLaunch then
        pcall(StopWallLaunch)
    end
    if RestoreForceFacing then
        pcall(RestoreForceFacing)
    end
    if SetCustomSkybox then
        pcall(function() SetCustomSkybox(false) end)
    end
end

-- One shared horizontal velocity limiter for all FHUB launch writers.
-- Vertical velocity is always preserved. Wall Momentum's own maximum only
-- applies while Wall Glide / Wall Momentum Carry is actively controlling it.
Settings._ClampHorizontalVelocity = function(velocity)
    local horizontal = Vector3.new(velocity.X, 0, velocity.Z)
    local cap = math.huge

    if Settings.SpeedCapper.Enabled then
        cap = math.min(cap, math.clamp(
            tonumber(Settings.SpeedCapper.Cap) or 500,
            Settings.SpeedCapper.Min,
            Settings.SpeedCapper.Max
        ))
    end

    if Settings.WallMomentum.Enabled
        and (WallGlideState.Active or WallGlideState.MomentumCarrying) then
        cap = math.min(cap, math.clamp(
            tonumber(Settings.WallMomentum.MaxSpeed) or 1200,
            Settings.WallMomentum.MaxSpeedMin,
            Settings.WallMomentum.MaxSpeedMax
        ))
    end

    if cap < math.huge and horizontal.Magnitude > cap and horizontal.Magnitude > 0.001 then
        horizontal = horizontal.Unit * cap
    end

    return Vector3.new(horizontal.X, velocity.Y, horizontal.Z)
end

local TextureOptimizationState = {
    Active = false,
    Properties = {},
    AddedConnection = nil,
}

local function SaveAndSet(instance, property, value)
    local values = TextureOptimizationState.Properties[instance]
    if not values then
        values = {}
        TextureOptimizationState.Properties[instance] = values
    end

    if values[property] == nil then
        local ok, original = pcall(function()
            return instance[property]
        end)
        if not ok then
            return false
        end
        values[property] = original
    end

    local ok = pcall(function()
        instance[property] = value
    end)
    return ok
end

local function RestoreTextureProperties()
    for instance, values in pairs(TextureOptimizationState.Properties) do
        if instance and instance.Parent then
            for property, original in pairs(values) do
                pcall(function()
                    instance[property] = original
                end)
            end
        end
    end
    table.clear(TextureOptimizationState.Properties)
end

local function OptimizeInstance(instance, optimization)
    if not instance or not instance.Parent then
        return
    end

    optimization = math.clamp(optimization or Settings.TextureOptimizer.Optimization, 0, 100)

    -- 25+: remove image-based texture layers.
    if optimization >= 25 and (instance:IsA("Texture") or instance:IsA("Decal")) then
        SaveAndSet(instance, "Transparency", 1)
        return
    end

    -- 25+: strip MeshPart texture detail where Roblox allows it.
    if optimization >= 25 and instance:IsA("MeshPart") then
        SaveAndSet(instance, "TextureID", "")
    end

    -- 50+: simplify geometry materials and reflections.
    if optimization >= 50 and instance:IsA("BasePart") then
        SaveAndSet(instance, "Material", Enum.Material.SmoothPlastic)
        SaveAndSet(instance, "Reflectance", 0)

        pcall(function()
            SaveAndSet(instance, "TopSurface", Enum.SurfaceType.Smooth)
        end)
        pcall(function()
            SaveAndSet(instance, "BottomSurface", Enum.SurfaceType.Smooth)
        end)
    end

    -- 75+: disable shadows and dynamic visual effects.
    if optimization >= 75 and instance:IsA("BasePart") then
        SaveAndSet(instance, "CastShadow", false)
    end

    if optimization >= 75
        and (instance:IsA("ParticleEmitter")
        or instance:IsA("Trail")
        or instance:IsA("Beam")) then
        SaveAndSet(instance, "Enabled", false)
    end
end

local function ApplyTextureOptimization(optimization)
    optimization = math.clamp(optimization or Settings.TextureOptimizer.Optimization, 0, 100)
    RestoreTextureProperties()
    TextureOptimizationState.Active = true

    if not TextureOptimizationState.AddedConnection then
        TextureOptimizationState.AddedConnection = workspace.DescendantAdded:Connect(function(instance)
            if TextureOptimizationState.Active then
                OptimizeInstance(instance, TextureOptimizationState.Level)
            end
        end)
    end

    TextureOptimizationState.Level = optimization

    for _, instance in ipairs(workspace:GetDescendants()) do
        OptimizeInstance(instance, optimization)
    end
end

RestoreTextureOptimization = function()
    TextureOptimizationState.Active = false
    TextureOptimizationState.Level = 0

    if TextureOptimizationState.AddedConnection then
        TextureOptimizationState.AddedConnection:Disconnect()
        TextureOptimizationState.AddedConnection = nil
    end

    RestoreTextureProperties()
end

local KARAMBIT_ASSET_ID = "rbxassetid://10744635356"
local KARAMBIT_TARGET_MAX_SIZE = 0.85

local function CSHandsSetPartProperties(part)
    if not part or not part:IsA("BasePart") then
        return
    end

    part.CanCollide = false
    part.CanTouch = false
    part.CanQuery = false
    part.CastShadow = false
    part.Massless = true
    part.Anchored = true
    part.TopSurface = Enum.SurfaceType.Smooth
    part.BottomSurface = Enum.SurfaceType.Smooth
end

local function CSHandsPrepareModel(root, partsTable)
    table.clear(partsTable)

    if root:IsA("BasePart") then
        CSHandsSetPartProperties(root)
        table.insert(partsTable, root)
        return
    end

    for _, object in ipairs(root:GetDescendants()) do
        if object:IsA("BasePart") then
            CSHandsSetPartProperties(object)
            table.insert(partsTable, object)
        end
    end
end

local function CSHandsLoadCustomModel(assetId, modelName, parent, partsTable)
    if not assetId or assetId == "" then
        return nil
    end

    local numericId = tostring(assetId):match("(%d+)")
    if not numericId then
        warn("[FortifiedHub] Invalid " .. modelName .. " asset ID")
        return nil
    end

    local ok, objects = pcall(function()
        return game:GetObjects("rbxassetid://" .. numericId)
    end)

    if not ok or not objects or #objects == 0 then
        warn("[FortifiedHub] Could not load custom " .. modelName .. " model: " .. numericId)
        return nil
    end

    local object = objects[1]
    object.Name = modelName

    for _, descendant in ipairs(object:GetDescendants()) do
        if descendant:IsA("Script")
            or descendant:IsA("LocalScript")
            or descendant:IsA("ModuleScript") then
            descendant:Destroy()
        end
    end

    object.Parent = parent
    CSHandsPrepareModel(object, partsTable)

    if #partsTable == 0 then
        object:Destroy()
        table.clear(partsTable)
        warn("[FortifiedHub] Custom " .. modelName .. " model has no BaseParts")
        return nil
    end

    return object
end

local function CSHandsHideOriginalArms(character)
    local names = {
        ["Left Arm"] = true,
        ["Right Arm"] = true,
        LeftUpperArm = true,
        LeftLowerArm = true,
        LeftHand = true,
        RightUpperArm = true,
        RightLowerArm = true,
        RightHand = true,
    }

    for _, object in ipairs(character:GetDescendants()) do
        if object:IsA("BasePart") and names[object.Name] then
            if CSHandsState.OriginalLTM[object] == nil then
                CSHandsState.OriginalLTM[object] = object.LocalTransparencyModifier
            end
            object.LocalTransparencyModifier = 1
        end
    end
end

local function CSHandsRestoreOriginalArms()
    for part, value in pairs(CSHandsState.OriginalLTM) do
        if part and part.Parent then
            pcall(function()
                part.LocalTransparencyModifier = value
            end)
        end
    end
    table.clear(CSHandsState.OriginalLTM)
end

local function CSHandsCollectKnifeParts(root)
    table.clear(CSHandsState.KarambitParts)

    if not root then
        return
    end

    if root:IsA("BasePart") then
        CSHandsSetPartProperties(root)
        table.insert(CSHandsState.KarambitParts, root)
        return
    end

    for _, obj in ipairs(root:GetDescendants()) do
        if obj:IsA("BasePart") then
            CSHandsSetPartProperties(obj)
            table.insert(CSHandsState.KarambitParts, obj)
        end
    end
end

local function CSHandsLoadKarambit(viewmodel)
    if CSHandsState.Karambit and CSHandsState.Karambit.Parent then
        CSHandsState.Karambit:Destroy()
    end

    CSHandsState.Karambit = nil
    table.clear(CSHandsState.KarambitParts)

    local ok, objects = pcall(function()
        return game:GetObjects(KARAMBIT_ASSET_ID)
    end)

    if not ok or not objects or #objects == 0 then
        warn("[FortifiedHub] CS POV Hands: could not load Karambit asset")
        return
    end

    local object = objects[1]
    object.Name = "Karambit"
    object.Parent = viewmodel
    CSHandsState.Karambit = object

    CSHandsCollectKnifeParts(object)

    if object:IsA("Model") then
        local okBox, _, size = pcall(function()
            local cf, extents = object:GetBoundingBox()
            return cf, extents
        end)

        if okBox and size and size.Magnitude > 0.01 then
            local scale = KARAMBIT_TARGET_MAX_SIZE / math.max(size.X, size.Y, size.Z)
            scale = math.clamp(scale, 0.75, 3.0)
            pcall(function()
                object:ScaleTo(scale)
            end)
        end
    end
end

local function CSHandsIsFirstPerson()
    local camera = workspace.CurrentCamera
    if not camera or not Character or not Character.Parent then
        return false
    end

    if Player.CameraMode == Enum.CameraMode.LockFirstPerson then
        return true
    end

    local head = Character:FindFirstChild("Head")
    if not head then
        return false
    end

    return (camera.CFrame.Position - head.Position).Magnitude <= 1.5
end

CSHandsDestroy = function()
    if CSHandsState.Connection then
        CSHandsState.Connection:Disconnect()
        CSHandsState.Connection = nil
    end

    if CSHandsState.CharacterConnection then
        CSHandsState.CharacterConnection:Disconnect()
        CSHandsState.CharacterConnection = nil
    end

    CSHandsRestoreOriginalArms()

    if CSHandsState.Karambit and CSHandsState.Karambit.Parent then
        pcall(function()
            CSHandsState.Karambit:Destroy()
        end)
    end

    CSHandsState.Karambit = nil
    table.clear(CSHandsState.KarambitParts)

    if CSHandsState.CustomHandsModel and CSHandsState.CustomHandsModel.Parent then
        pcall(function() CSHandsState.CustomHandsModel:Destroy() end)
    end
    CSHandsState.CustomHandsModel = nil
    table.clear(CSHandsState.CustomHandsParts)

    if CSHandsState.CustomWeaponModel and CSHandsState.CustomWeaponModel.Parent then
        pcall(function() CSHandsState.CustomWeaponModel:Destroy() end)
    end
    CSHandsState.CustomWeaponModel = nil
    table.clear(CSHandsState.CustomWeaponParts)

    if CSHandsState.Viewmodel then
        pcall(function()
            CSHandsState.Viewmodel:Destroy()
        end)
        CSHandsState.Viewmodel = nil
    end
end

        rightArm.Size = Vector3.new(0.48, 2.75, 0.48)
        CSHandsSetPartProperties(rightArm)
        rightArm.Parent = viewmodel

        local charLeft = character:FindFirstChild("Left Arm")
            or character:FindFirstChild("LeftUpperArm")
            or character:FindFirstChild("LeftHand")
        local charRight = character:FindFirstChild("Right Arm")
            or character:FindFirstChild("RightUpperArm")
            or character:FindFirstChild("RightHand")

            leftArm.Color = charLeft.Color
            leftArm.Material = charLeft.Material
        else
            leftArm.Color = Color3.fromRGB(255, 204, 153)
        end

        if charRight and charRight:IsA("BasePart") then
            rightArm.Color = charRight.Color
            rightArm.Material = charRight.Material
        else
            rightArm.Color = Color3.fromRGB(255, 204, 153)
        end

        if shirt then
            local viewShirt = Instance.new("Shirt")
            viewShirt.Name = "CSHandsShirt"
            viewShirt.ShirtTemplate = shirt.ShirtTemplate
            viewShirt.Parent = viewmodel
        end
    end

    if Settings.Visuals.CustomWeaponModelId ~= "" then
        CSHandsState.CustomWeaponModel = CSHandsLoadCustomModel(
            Settings.Visuals.CustomWeaponModelId,
            "CustomWeapon",
            viewmodel,
            CSHandsState.CustomWeaponParts
        )
    end

    if not CSHandsState.CustomWeaponModel then
        CSHandsLoadKarambit(viewmodel)
    end

    CSHandsHideOriginalArms(character)

    CSHandsState.Connection = RunService.RenderStepped:Connect(function()
        if not Settings.Visuals.CSHands
            or not Character
            or not Character.Parent
            or not CSHandsState.Viewmodel
            or CSHandsState.Viewmodel ~= viewmodel then
            return
        end

        local currentCamera = workspace.CurrentCamera
        if not currentCamera then
            return
        end

        CSHandsHideOriginalArms(Character)

        local visible = CSHandsIsFirstPerson()
        local alpha = visible and 0 or 1

        if leftArm then
            leftArm.LocalTransparencyModifier = alpha
        end
        if rightArm then
            rightArm.LocalTransparencyModifier = alpha
        end

        for _, part in ipairs(CSHandsState.CustomHandsParts) do
            if part and part.Parent then
                part.LocalTransparencyModifier = alpha
            end
        end

        for _, part in ipairs(CSHandsState.KarambitParts) do
            if part and part.Parent then
                part.LocalTransparencyModifier = alpha
            end
        end

        for _, part in ipairs(CSHandsState.CustomWeaponParts) do
            if part and part.Parent then
                part.LocalTransparencyModifier = alpha
            end
        end

        if not visible then
            return
        end

        -- Fully customizable camera-relative pose.
        local vmRoot = currentCamera.CFrame * CFrame.new(
            Settings.Visuals.HandPosition.X,
            Settings.Visuals.HandPosition.Y,
            Settings.Visuals.HandPosition.Z
        )

        if CSHandsState.CustomHandsModel then
            local handsCF = vmRoot
                * CFrame.new(Settings.Visuals.RightHandOffset.X, Settings.Visuals.RightHandOffset.Y, Settings.Visuals.RightHandOffset.Z)
                * CFrame.Angles(
                    math.rad(Settings.Visuals.HandRotation.X),
                    math.rad(Settings.Visuals.HandRotation.Y),
                    math.rad(Settings.Visuals.HandRotation.Z)
                )
            pcall(function()
                if Settings.Visuals.HandScale ~= 1 then
                    CSHandsState.CustomHandsModel:ScaleTo(Settings.Visuals.HandScale)
                end
                CSHandsState.CustomHandsModel:PivotTo(handsCF)
            end)
        else
            local handRotLeft = CFrame.Angles(
                math.rad(Settings.Visuals.HandRotation.X),
                math.rad(-math.abs(Settings.Visuals.HandRotation.Y)),
                math.rad(-math.abs(Settings.Visuals.HandRotation.Z))
            )
            local handRotRight = CFrame.Angles(
                math.rad(Settings.Visuals.HandRotation.X),
                math.rad(math.abs(Settings.Visuals.HandRotation.Y)),
                math.rad(math.abs(Settings.Visuals.HandRotation.Z))
            )

            local handScale = math.max(Settings.Visuals.HandScale, 0.1)
            leftArm.Size = Vector3.new(0.48, 2.75, 0.48) * handScale
            rightArm.Size = Vector3.new(0.48, 2.75, 0.48) * handScale

            leftArm.CFrame = vmRoot
                * CFrame.new(Settings.Visuals.LeftHandOffset.X, Settings.Visuals.LeftHandOffset.Y, Settings.Visuals.LeftHandOffset.Z)
                * handRotLeft

            rightArm.CFrame = vmRoot
                * CFrame.new(Settings.Visuals.RightHandOffset.X, Settings.Visuals.RightHandOffset.Y, Settings.Visuals.RightHandOffset.Z)
                * handRotRight
        end

        local attachCF
        if rightArm then
            attachCF = rightArm.CFrame
        else
            attachCF = vmRoot
                * CFrame.new(Settings.Visuals.RightHandOffset.X, Settings.Visuals.RightHandOffset.Y, Settings.Visuals.RightHandOffset.Z)
                * CFrame.Angles(
                    math.rad(Settings.Visuals.HandRotation.X),
                    math.rad(Settings.Visuals.HandRotation.Y),
                    math.rad(Settings.Visuals.HandRotation.Z)
                )
        end

        local knifeBaseCF = attachCF
            * CFrame.new(Settings.Visuals.KnifeOffset.X, Settings.Visuals.KnifeOffset.Y, Settings.Visuals.KnifeOffset.Z)
            * CFrame.Angles(
                math.rad(Settings.Visuals.KnifeRotation.X),
                math.rad(Settings.Visuals.KnifeRotation.Y),
                math.rad(Settings.Visuals.KnifeRotation.Z)
            )

        local weaponModel = CSHandsState.CustomWeaponModel or CSHandsState.Karambit
        if weaponModel then
            pcall(function()
                if Settings.Visuals.KnifeScale ~= 1 and weaponModel:IsA("Model") then
                    weaponModel:ScaleTo(Settings.Visuals.KnifeScale)
                end
                if weaponModel:IsA("Model") then
                    weaponModel:PivotTo(knifeBaseCF)
                else
                    local parts = CSHandsState.CustomWeaponModel and CSHandsState.CustomWeaponParts or CSHandsState.KarambitParts
                    local basePart = parts[1]
                    if basePart and basePart.Parent then
                        basePart.CFrame = knifeBaseCF
                    end
                end
            end)
        end
    end)
end

SetFogRemoved = function(enabled)

        if FogState.Saved then

        FogState.FogEnd = Lighting.FogEnd
        table.clear(FogState.Atmospheres)

        Lighting.FogStart = 0
        Lighting.FogEnd = 1000000

        for _, instance in ipairs(Lighting:GetChildren()) do
            if instance:IsA("Atmosphere") then
                FogState.Atmospheres[instance] = {
                    Density = instance.Density,
                    Offset = instance.Offset,
                    Haze = instance.Haze,
                    Glare = instance.Glare,
                }

                instance.Density = 0
                instance.Offset = 0
                instance.Haze = 0
                instance.Glare = 0
            end
        end
    else
        if not FogState.Saved then
            return
        end

        Lighting.FogStart = FogState.FogStart
        Lighting.FogEnd = FogState.FogEnd

                atmosphere.Offset = values.Offset

    end

local SkyboxState = {
    Saved = false,
    Originals = {},
    CustomSky = nil,

    -- Keep custom skybox visual-only so swapping sky textures does not
    -- change the world's environment-derived lighting/reflections.
    OriginalEnvironmentDiffuseScale = nil,
    OriginalEnvironmentSpecularScale = nil,
}

local function NormalizeAssetId(value)
    local id = tostring(value or ""):match("%d+")
    if not id or id == "" then
        return nil
    end
    return "rbxassetid://" .. id
end

SetCustomSkybox = function(enabled)
    if enabled then
        local assetId = NormalizeAssetId(Settings.Visuals.SkyboxId)

        if not assetId then
            return false, "Enter a skybox asset ID."
        end

        -- Reuse the current custom Sky when the feature is already active.
        if SkyboxState.Saved and SkyboxState.CustomSky then
            local ok = pcall(function()
                SkyboxState.CustomSky.SkyboxBk = assetId
                SkyboxState.CustomSky.SkyboxDn = assetId
                SkyboxState.CustomSky.SkyboxFt = assetId
                SkyboxState.CustomSky.SkyboxLf = assetId
                SkyboxState.CustomSky.SkyboxRt = assetId
                SkyboxState.CustomSky.SkyboxUp = assetId
            end)

            if ok then
                return true, "Skybox updated."
            end
        end

        if not SkyboxState.Saved then
            SkyboxState.Saved = true
            table.clear(SkyboxState.Originals)

            -- Roblox can derive ambient/specular contribution from the
            -- environment. Save those values and temporarily disable them
            -- so changing the sky texture does not brighten/darken the POV.
            SkyboxState.OriginalEnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale
            SkyboxState.OriginalEnvironmentSpecularScale = Lighting.EnvironmentSpecularScale
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0

            for _, instance in ipairs(Lighting:GetChildren()) do
                if instance:IsA("Sky") then
                    table.insert(SkyboxState.Originals, instance)
                    instance.Parent = nil
                end
            end
        end

        local sky = SkyboxState.CustomSky
        if not sky then
            sky = Instance.new("Sky")
            sky.Name = "FortifiedHubCustomSkybox"
            SkyboxState.CustomSky = sky
        end

        local ok = pcall(function()
            sky.SkyboxBk = assetId
            sky.SkyboxDn = assetId
            sky.SkyboxFt = assetId
            sky.SkyboxLf = assetId
            sky.SkyboxRt = assetId
            sky.SkyboxUp = assetId
            sky.Parent = Lighting
        end)

        if not ok then
            sky:Destroy()
            SkyboxState.CustomSky = nil
            return false, "Could not apply the supplied skybox asset ID."
        end

        return true, "Custom skybox enabled on all six faces."
    end

    if not SkyboxState.Saved then
        return true, "Custom skybox disabled."
    end

    if SkyboxState.CustomSky and SkyboxState.CustomSky.Parent then
        SkyboxState.CustomSky:Destroy()
    end
    SkyboxState.CustomSky = nil

    for _, sky in ipairs(SkyboxState.Originals) do
        if sky and sky.Parent == nil then
            sky.Parent = Lighting
        end
    end

    -- Restore the exact environment-lighting values from before the custom
    -- skybox was enabled.
    if SkyboxState.OriginalEnvironmentDiffuseScale ~= nil then
        Lighting.EnvironmentDiffuseScale = SkyboxState.OriginalEnvironmentDiffuseScale
    end
    if SkyboxState.OriginalEnvironmentSpecularScale ~= nil then
        Lighting.EnvironmentSpecularScale = SkyboxState.OriginalEnvironmentSpecularScale
    end

    SkyboxState.OriginalEnvironmentDiffuseScale = nil
    SkyboxState.OriginalEnvironmentSpecularScale = nil
    table.clear(SkyboxState.Originals)
    SkyboxState.Saved = false
    return true, "Original skybox restored."
end

local COLORS

local function UpdateStrafeOptimizerStatus()
    if StrafeOptimizerStatusLabel and StrafeOptimizerStatusLabel.Parent then
        local effective = Settings.StrafeOptimizer.Sensitivity * StrafeOptimizerState.AutoMultiplier
        local statusText = StrafeOptimizerState.AutoTuneStatus
        if StrafeOptimizerState.AutoTuneRunning then
            local elapsed = math.max(0, os.clock() - StrafeOptimizerState.AutoTuneStartedAt)
            local remaining = math.max(0, Settings.StrafeOptimizer.AutoTuneDuration - elapsed)
            statusText = string.format("TUNING %.1fs", remaining)
        end
        StrafeOptimizerStatusLabel.Text = string.format("Effective: %.2fx  |  %s", effective, statusText)
    end

    -- Green means calibration is actively running. Every non-running state
    -- (ready, completed, or no input) is red, as requested.
    if StrafeOptimizerAutoTuneButton and StrafeOptimizerAutoTuneButton.Parent then
        if StrafeOptimizerState.AutoTuneRunning then
            StrafeOptimizerAutoTuneButton.Text = "Tuning... move mouse"
            StrafeOptimizerAutoTuneButton.BackgroundColor3 = COLORS.AutoTuneOn
        elseif StrafeOptimizerState.AutoTuneStatus == "BALANCED" then
            StrafeOptimizerAutoTuneButton.Text = "AutoTune Again"
            StrafeOptimizerAutoTuneButton.BackgroundColor3 = COLORS.AutoTuneOff
        elseif StrafeOptimizerState.AutoTuneStatus == "NO INPUT" then
            StrafeOptimizerAutoTuneButton.Text = "Retry AutoTune"
            StrafeOptimizerAutoTuneButton.BackgroundColor3 = COLORS.AutoTuneOff
        else
            StrafeOptimizerAutoTuneButton.Text = "AutoTune • Balance"
            StrafeOptimizerAutoTuneButton.BackgroundColor3 = COLORS.AutoTuneOff
        end
    end
end

local function ApplyStrafeOptimizerSensitivity()
    if not StrafeOptimizerState.Active or StrafeOptimizerState.OriginalSensitivity == nil then
        return
    end

    local manualMultiplier = math.clamp(
        Settings.StrafeOptimizer.Sensitivity,
        Settings.StrafeOptimizer.SensitivityMin,
        Settings.StrafeOptimizer.SensitivityMax
    )
    local autoMultiplier = StrafeOptimizerState.AutoMultiplier or 1
    local target = StrafeOptimizerState.OriginalSensitivity * manualMultiplier * autoMultiplier
    target = math.clamp(target, 0, 10)

    pcall(function()
        UIS.MouseDeltaSensitivity = target
    end)
end

RestoreStrafeOptimizerSensitivity = function()
    if StrafeOptimizerState.Active and StrafeOptimizerState.OriginalSensitivity ~= nil then
        pcall(function()
            UIS.MouseDeltaSensitivity = StrafeOptimizerState.OriginalSensitivity
        end)
    end

    StrafeOptimizerState.Active = false
    StrafeOptimizerState.OriginalSensitivity = nil
    StrafeOptimizerState.AutoMultiplier = 1
    StrafeOptimizerState.PreviousLookVector = nil
    StrafeOptimizerState.AutoTuneRunning = false
    StrafeOptimizerState.AutoTuneStartedAt = 0
    StrafeOptimizerState.AutoTuneStartingMultiplier = 1
    StrafeOptimizerState.AutoTuneSawTurn = false
    StrafeOptimizerState.LastStatusUpdateAt = 0
    table.clear(StrafeOptimizerState.TurnRateSamples)
    StrafeOptimizerState.AutoTuneStatus = "READY"
    UpdateStrafeOptimizerStatus()
end

local function SetStrafeOptimizerEnabled(enabled)
    Settings.StrafeOptimizer.Enabled = enabled

    if enabled then
        if not StrafeOptimizerState.Active then
            local ok, current = pcall(function()
                return UIS.MouseDeltaSensitivity
            end)
            if not ok or type(current) ~= "number" then
                Settings.StrafeOptimizer.Enabled = false
                warn("[FortifiedHub] Strafe Optimizer could not read mouse sensitivity on this client.")
                return
            end

            StrafeOptimizerState.OriginalSensitivity = current
            StrafeOptimizerState.Active = true
            StrafeOptimizerState.AutoMultiplier = 1
            StrafeOptimizerState.PreviousLookVector = nil
            StrafeOptimizerState.AutoTuneRunning = false
            StrafeOptimizerState.AutoTuneStartedAt = 0
            StrafeOptimizerState.AutoTuneStartingMultiplier = 1
            StrafeOptimizerState.AutoTuneSawTurn = false
            StrafeOptimizerState.LastStatusUpdateAt = 0
            table.clear(StrafeOptimizerState.TurnRateSamples)
            StrafeOptimizerState.AutoTuneStatus = "READY"
        end
        ApplyStrafeOptimizerSensitivity()
    else
        RestoreStrafeOptimizerSensitivity()
    end

    UpdateStrafeOptimizerStatus()
end

-- AutoTune is a one-shot action. It collects camera-turn samples at the
-- current sensitivity for the full window, then applies one bounded correction.
local function BeginStrafeAutoTune()
    if not Settings.StrafeOptimizer.Enabled then
        SetStrafeOptimizerEnabled(true)
    end

    if not StrafeOptimizerState.Active then
        StrafeOptimizerState.AutoTuneRunning = false
        StrafeOptimizerState.AutoTuneStatus = "NO INPUT"
        UpdateStrafeOptimizerStatus()
        return
    end

    local camera = workspace.CurrentCamera
    -- Keep the previous result while measuring. If AutoTune is run again,
    -- the next correction is calculated relative to that currently applied value.
    StrafeOptimizerState.AutoTuneStartingMultiplier = StrafeOptimizerState.AutoMultiplier or 1
    StrafeOptimizerState.AutoTuneRunning = true
    StrafeOptimizerState.AutoTuneStartedAt = os.clock()
    StrafeOptimizerState.AutoTuneSawTurn = false
    StrafeOptimizerState.LastStatusUpdateAt = 0
    table.clear(StrafeOptimizerState.TurnRateSamples)
    StrafeOptimizerState.PreviousLookVector = camera and camera.CFrame.LookVector or nil
    StrafeOptimizerState.AutoTuneStatus = "TUNING"

    ApplyStrafeOptimizerSensitivity()
    UpdateStrafeOptimizerStatus()
end

local function CreateSlider(parent, labelText, minValue, maxValue, startValue, callback, y, step)
    step = step or 1

    local function formatValue(value)
        if step < 1 then
            return string.format("%.1f", value)
        end
        return tostring(math.floor(value + 0.5))
    end

    local Holder = Instance.new("Frame")
    Holder.Size = UDim2.new(1, -28, 0, 50)
    Holder.Position = UDim2.fromOffset(14, y)
    Holder.BackgroundTransparency = 1
    Holder.Parent = parent

    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(1, 0, 0, 18)
    Label.Position = UDim2.fromOffset(0, 0)
    Label.BackgroundTransparency = 1
    Label.Text = labelText .. ": " .. formatValue(startValue)
    Label.TextColor3 = Color3.fromRGB(255, 255, 255)
    Label.TextSize = 11
    Label.Font = Enum.Font.GothamBold
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = Holder

    local Bar = Instance.new("Frame")
    Bar.Size = UDim2.new(1, -2, 0, 8)
    Bar.Position = UDim2.fromOffset(1, 27)
    Bar.BackgroundColor3 = COLORS.Slider
    Bar.BorderSizePixel = 0
    Bar.Active = true
    Bar.Parent = Holder
    Corner(Bar, 4)

    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new(0, 0, 1, 0)
    Fill.BackgroundColor3 = COLORS.SliderFill
    Fill.BorderSizePixel = 0
    Fill.Parent = Bar
    Corner(Fill, 4)

    local Knob = Instance.new("TextButton")
    Knob.Size = UDim2.fromOffset(14, 14)
    Knob.AnchorPoint = Vector2.new(0.5, 0.5)
    Knob.BackgroundColor3 = Color3.fromRGB(224, 252, 255)
    Knob.BorderSizePixel = 0
    Knob.Text = ""
    Knob.AutoButtonColor = false
    Knob.Parent = Bar
    Corner(Knob, 7)

    local dragging = false

    local function SetValue(value, invokeCallback)
        value = math.clamp(
            math.floor((tonumber(value) / step) + 0.5) * step,
            minValue,
            maxValue
        )

        local normalized =
            (value - minValue) /
            math.max(maxValue - minValue, step)

        Fill.Size = UDim2.new(normalized, 0, 1, 0)
        Knob.Position = UDim2.new(normalized, 0, 0.5, 0)
        Label.Text = labelText .. ": " .. formatValue(value)

        if invokeCallback ~= false then
            callback(value)
        end

        return value
    end

    local function SetFromX(x)
        local left = Bar.AbsolutePosition.X
        local width = Bar.AbsoluteSize.X

        if width <= 0 then
            return
        end

        local percent = math.clamp((x - left) / width, 0, 1)
        local rawValue = minValue + ((maxValue - minValue) * percent)
        SetValue(rawValue, true)
    end

    -- Initialize from the numeric value directly; this also works before
    -- Roblox has completed the first layout pass and assigned AbsoluteSize.
    SetValue(startValue, true)

    Connect(Knob.MouseButton1Down, function()
        dragging = true
    end)

    Connect(Bar.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            SetFromX(input.Position.X)
        end
    end)

    Connect(UIS.InputChanged, function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            SetFromX(input.Position.X)
        end
    end)

    Connect(UIS.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

    return Holder, function(value)
        return SetValue(value, true)
    end
end

local function BindingName(binding)
    if not binding then
        return "Unbound"
    end

    if binding.Type == Enum.UserInputType.Keyboard then
        if binding.Code == Enum.KeyCode.F13 then
            return "Mouse Side 1"
        elseif binding.Code == Enum.KeyCode.F14 then
            return "Mouse Side 2"
        end
        return binding.Code.Name
    end

    if binding.Type == Enum.UserInputType.MouseButton1 then
        return "Button 1"
    elseif binding.Type == Enum.UserInputType.MouseButton2 then
        return "Button 2"
    elseif binding.Type == Enum.UserInputType.MouseButton3 then
        return "Button 3"
    end

    return "Unbound"
end

local function InputMatches(binding, input)
    if not binding then
        return false
    end

    if binding.Type == Enum.UserInputType.Keyboard then
        return input.UserInputType == Enum.UserInputType.Keyboard
            and input.KeyCode == binding.Code
    end

    return input.UserInputType == binding.Type
end

-- Poll the actual physical state for Spin Bot instead of relying only on
-- InputBegan/InputEnded. This makes the custom bind reliable even when the
-- game marks the input as processed (for example LeftShift being used by
-- another action). W/S are still rejected by the bind setter.
local function IsSpinBindingHeld(binding)
    if not binding then
        return false
    end

    if binding.Type == Enum.UserInputType.Keyboard then
        return UIS:IsKeyDown(binding.Code)
    end

    if binding.Type == Enum.UserInputType.MouseButton1
        or binding.Type == Enum.UserInputType.MouseButton2
        or binding.Type == Enum.UserInputType.MouseButton3 then
        return UIS:IsMouseButtonPressed(binding.Type)
    end

    return false
end

-- Check whether any configured keyboard/mouse binding is physically held.
local function IsBindingHeld(binding)
    if not binding then
        return false
    end

    if binding.Type == Enum.UserInputType.Keyboard then
        return UIS:IsKeyDown(binding.Code)
    end

    if binding.Type == Enum.UserInputType.MouseButton1
        or binding.Type == Enum.UserInputType.MouseButton2
        or binding.Type == Enum.UserInputType.MouseButton3 then
        return UIS:IsMouseButtonPressed(binding.Type)
    end

    return false
end

local Rebinding = false

local function CreateKeyButton(parent, binding, callback)
    local Button = Instance.new("TextButton")
    Button.Size = UDim2.fromOffset(96, 26)
    Button.AnchorPoint = Vector2.new(1, 0)
    Button.Position = UDim2.new(1, -12, 0, 48)
    Button.BackgroundColor3 = Color3.fromRGB(8, 27, 39)
    Button.BorderSizePixel = 0
    Button.Text = BindingName(binding)
    Button.TextColor3 = Color3.fromRGB(255, 255, 255)
    Button.TextSize = 11
    Button.Font = Enum.Font.GothamBold
    Button.AutoButtonColor = false
    Button.Parent = parent
    Corner(Button, 8)

    local waiting = false

    Connect(Button.MouseButton1Click, function()
        if waiting then
            return
        end

        waiting = true
        Rebinding = true
        Button.Text = "Press key..."

        local connection

        connection = Connect(UIS.InputBegan, function(input)
            local valid =
                input.UserInputType == Enum.UserInputType.Keyboard
                or input.UserInputType == Enum.UserInputType.MouseButton1
                or input.UserInputType == Enum.UserInputType.MouseButton2
                or input.UserInputType == Enum.UserInputType.MouseButton3

            if not valid then
                return
            end

            local newBinding

            if input.UserInputType == Enum.UserInputType.Keyboard then
                newBinding = {
                    Type = Enum.UserInputType.Keyboard,
                    Code = input.KeyCode,
                }
            else
                newBinding = {
                    Type = input.UserInputType,
                    Code = input.UserInputType,
                }
            end

            local accepted = callback(newBinding)
            if accepted ~= false then
                Button.Text = BindingName(newBinding)
            end
            waiting = false
            connection:Disconnect()

            task.defer(function()
                Rebinding = false
            end)
        end)
    end)

    return Button
end

local function AddCard(grid, config)
    local height = config.Height or 232
    local gridLayout = grid:FindFirstChildOfClass("UIGridLayout")
    if gridLayout then
        local sharedHeight = math.max(GridCardHeights[grid] or 232, height)
        GridCardHeights[grid] = sharedHeight
        gridLayout.CellSize = UDim2.fromOffset(CARD_WIDTH, sharedHeight)
    end

    local Card = Instance.new("Frame")
    Card.Name = config.Name or "Feature"
    Card.Size = UDim2.fromOffset(CARD_WIDTH, height)
    Card.BackgroundColor3 = COLORS.Card
    Card.BorderSizePixel = 0
    Card.LayoutOrder = config.Order or 1
    Card.Parent = grid
    Corner(Card, 15)

    local CardGradient = Instance.new("UIGradient")
    CardGradient.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(12, 38, 52)),
        ColorSequenceKeypoint.new(1, COLORS.Card),
    })
    CardGradient.Rotation = 90
    CardGradient.Parent = Card

    local Stroke = Instance.new("UIStroke")
    Stroke.Color = COLORS.Border
    Stroke.Thickness = 1
    Stroke.Transparency = 0.18
    Stroke.Parent = Card

    local Title = Instance.new("TextLabel")
    Title.Size = UDim2.new(1, -104, 0, 24)
    Title.Position = UDim2.fromOffset(14, 10)
    Title.BackgroundTransparency = 1
    Title.Text = config.Title
    Title.TextColor3 = COLORS.Text
    Title.TextSize = 15
    Title.Font = Enum.Font.GothamBold
    Title.TextXAlignment = Enum.TextXAlignment.Left
    Title.Parent = Card

    local Desc = Instance.new("TextLabel")
    local descriptionRightOffset = config.ReserveKeyArea and -146
        or (config.Key and -132 or -28)
    Desc.Size = UDim2.new(1, descriptionRightOffset, 0, 48)
    Desc.Position = UDim2.fromOffset(14, 40)
    Desc.BackgroundTransparency = 1
    Desc.Text = config.Description or ""
    Desc.TextColor3 = Color3.fromRGB(205, 222, 230)
    Desc.TextSize = 10
    Desc.Font = Enum.Font.Gotham
    Desc.TextWrapped = true
    Desc.TextXAlignment = Enum.TextXAlignment.Left
    Desc.TextYAlignment = Enum.TextYAlignment.Top
    Desc.Parent = Card

    local toggleButton
    local setToggle
    local syncToggle

    if config.Toggle then
        toggleButton, setToggle = CreatePill(Card)

        local function SyncToggle()
            setToggle(config.GetEnabled and config.GetEnabled() or false)
        end
        syncToggle = SyncToggle

        SyncToggle()

        Connect(toggleButton.MouseButton1Click, function()
            local current = config.GetEnabled and config.GetEnabled() or false
            if config.OnToggle then
                config.OnToggle(not current)
            end
            SyncToggle()
        end)
    end

    if config.Key then
        CreateKeyButton(
            Card,
            config.GetKey and config.GetKey() or config.Key,
            config.OnKey
        )
    end

    local sliderCount = config.Sliders and #config.Sliders or 0
    local sliderSetters = {}

    if sliderCount > 0 then
        for i, slider in ipairs(config.Sliders) do
            local sliderY

            if sliderCount == 1 then
                sliderY = 92
            else
                sliderY = 90 + ((i - 1) * 48)
            end

            local _, setSliderValue = CreateSlider(
                Card,
                slider.Name,
                slider.Min,
                slider.Max,
                slider.GetValue(),
                slider.SetValue,
                sliderY,
                slider.Step
            )

            sliderSetters[slider.Name] = setSliderValue
        end
    end

    if config.NumericInputs then
        for _, numeric in ipairs(config.NumericInputs) do
            local syncSlider = sliderSetters[numeric.SliderName]
            CreateNumberInput(
                Card,
                numeric.Name,
                numeric.GetValue(),
                numeric.Min,
                numeric.Max,
                function(value)
                    if syncSlider then
                        syncSlider(value)
                    else
                        numeric.SetValue(value)
                    end
                end,
                numeric.Y
            )
        end
    end

    if config.ActionButtons then
        for _, action in ipairs(config.ActionButtons) do
            local feedbackToken = 0
            local actionButton
            actionButton = CreateSmallButton(Card, action.Text, action.Y, function()
                local result = action.Callback()
                if type(result) == "string" and result ~= "" then
                    feedbackToken += 1
                    local thisToken = feedbackToken
                    actionButton.Text = result
                    task.delay(action.FeedbackDuration or 0.9, function()
                        if actionButton and actionButton.Parent and feedbackToken == thisToken then
                            actionButton.Text = action.Text
                        end
                    end)
                end
            end)
        end
    end

    return Card, syncToggle
end

local function ConnectPageCanvas(page)
    local layout = Instance.new("UIListLayout")
    layout.Padding = UDim.new(0, 10)
    layout.SortOrder = Enum.SortOrder.LayoutOrder
    layout.Parent = page

    Connect(layout:GetPropertyChangedSignal("AbsoluteContentSize"), function()
        page.CanvasSize = UDim2.fromOffset(
            0,
            layout.AbsoluteContentSize.Y + 20
        )
    end)

    task.defer(function()
        page.CanvasSize = UDim2.fromOffset(
            0,
            layout.AbsoluteContentSize.Y + 20
        )
    end)
end

        if not Root or not Root.Parent then

            return false

    local function ReleaseStoredImpulse()

        if not Settings.ImpulseStorage.Enabled
            or not Root or not Root.Parent
            or not Humanoid or Humanoid.Health <= 0
            or Runtime.ImpulseCharge < 5
            or now - Runtime.LastImpulseRelease < 0.15 then
            return false
        end
        if not CanStartBurst("ImpulseStorage", 0.22) then
            return false
        end

        local charge = math.clamp(Runtime.ImpulseCharge, 0, Settings.ImpulseStorage.MaxCharge)
        local multiplier = charge / 100
        local power = Settings.ImpulseStorage.ReleasePower * multiplier
        local current = Root.AssemblyLinearVelocity
        local direction = GetDirection(Settings.ImpulseStorage.DirectionMode)
        if direction.Magnitude < 0.01 then
            ReleaseBurst("ImpulseStorage")
            return false
        end

        local horizontal = Flat(current) + direction * power
        CommitVelocity(Vector3.new(horizontal.X, current.Y, horizontal.Z))
        Runtime.ImpulseCharge = 0
        Runtime.LastImpulseRelease = now
        return true
    end

    local function StartSpiralLaunch()
        local now = os.clock()
        local config = Settings.SpiralLaunch
        if not config.Enabled or Runtime.SpiralActive
            or not Root or not Root.Parent
            or not Humanoid or Humanoid.Health <= 0
            or now - Runtime.SpiralLastLaunch < config.Cooldown then
            return false
        end
        if not CanStartBurst("SpiralLaunch", config.Duration + 0.15) then
            return false
        end

        local current = Root.AssemblyLinearVelocity
        local direction = GetDirection("Momentum")
        if direction.Magnitude < 0.01 then
            direction = GetDirection("Camera")
        end
        if direction.Magnitude < 0.01 then
            ReleaseBurst("SpiralLaunch")
            return false
        end

        Runtime.SpiralDirection = direction
        local requestedHorizontalSpeed = math.max(Flat(current).Magnitude, config.SpiralPower)
        if Settings.SpeedCapper.Enabled then
            requestedHorizontalSpeed = math.min(
                requestedHorizontalSpeed,
                math.clamp(Settings.SpeedCapper.Cap, Settings.SpeedCapper.Min, Settings.SpeedCapper.Max)
            )
        end
        Runtime.SpiralHorizontalSpeed = requestedHorizontalSpeed
        Runtime.SpiralStartedAt = now
        Runtime.SpiralLastLaunch = now
        Runtime.SpiralActive = true

        local maxHorizontal = Runtime.SpiralHorizontalSpeed
        local launchVelocity = Vector3.new(
            direction.X * maxHorizontal,
            math.max(current.Y, 0) + config.VerticalPower,
            direction.Z * maxHorizontal
        )
        CommitVelocity(launchVelocity)
        return true
    end

    Settings._PropulsionUpdate = function(deltaTime)
        local now = os.clock()

        if not Root or not Root.Parent or not Humanoid or Humanoid.Health <= 0 then
            Runtime.ImpulseCharge = 0
            Runtime.SpiralActive = false
            ReleaseBurst("SpiralLaunch")
            return
        end

        -- IMPULSE STORAGE: charge while moving at or above the threshold.
        if Settings.ImpulseStorage.Enabled then
            local horizontalSpeed = Flat(Root.AssemblyLinearVelocity).Magnitude
            local threshold = math.max(1, Settings.ImpulseStorage.ThresholdSpeed)
            if horizontalSpeed >= threshold then
                local speedFactor = math.clamp(horizontalSpeed / threshold, 1, 3)
                Runtime.ImpulseCharge = math.min(
                    Settings.ImpulseStorage.MaxCharge,
                    Runtime.ImpulseCharge + Settings.ImpulseStorage.ChargeRate * speedFactor * dt
                )
            end
            -- Keep charge while the player briefly slows down; disabling the
            -- feature or respawning still clears it. This makes the stored
            -- impulse genuinely releasable after a short speed fluctuation.
        else
            Runtime.ImpulseCharge = 0
        end

        local maxCharge = math.max(1, Settings.ImpulseStorage.MaxCharge)
        Runtime.ImpulseCharge = math.clamp(Runtime.ImpulseCharge, 0, maxCharge)
        chargeLabel.Text = string.format(
            "CHARGE: %d / %d%%",
            math.floor(Runtime.ImpulseCharge + 0.5),
            math.floor(maxCharge + 0.5)
        )
        chargeFill.Size = UDim2.new(math.clamp(Runtime.ImpulseCharge / maxCharge, 0, 1), 0, 1, 0)

        -- CORNER POP: confirm a wall beside the player and verify the wall
        -- does not continue a short distance ahead along the movement tangent.
        if Settings.CornerPop.Enabled then
            local horizontalVelocity = Flat(Root.AssemblyLinearVelocity)
            local moveDirection = Flat(Humanoid.MoveDirection)
            if moveDirection.Magnitude < 0.05 and horizontalVelocity.Magnitude > 0.05 then
                moveDirection = horizontalVelocity.Unit
            end

            local cornerInfo = nil
            if moveDirection.Magnitude > 0.05 and horizontalVelocity.Magnitude >= 35 then
                moveDirection = moveDirection.Unit
                local wall = FindWall(Settings.CornerPop.DetectionDistance)

                if wall then
                    local normal = Flat(wall.Normal)
                    if normal.Magnitude > 0.05 then
                        normal = normal.Unit
                        local normalDot = moveDirection:Dot(normal)
                        local tangent = moveDirection - normal * normalDot
                        if math.abs(normalDot) <= 0.72 and tangent.Magnitude > 0.35 then
                            tangent = tangent.Unit
                            local localNormal = Root.CFrame:VectorToObjectSpace(normal)
                            local halfExtent =
                                math.abs(localNormal.X) * (Root.Size.X * 0.5)
                                + math.abs(localNormal.Z) * (Root.Size.Z * 0.5)
                            local reach = halfExtent + Settings.CornerPop.DetectionDistance + 0.8
                            local params = RaycastParams.new()
                            params.FilterType = Enum.RaycastFilterType.Exclude
                            params.FilterDescendantsInstances = {Character}
                            pcall(function() params.RespectCanCollide = true end)

                            local currentOrigin = Root.Position + normal * reach
                            local currentWall = workspace:Raycast(currentOrigin, -normal * (reach + 0.5), params)
                            local aheadOrigin = Root.Position + tangent * math.max(2.5, Root.Size.Z * 1.4) + normal * reach
                            local wallAhead = workspace:Raycast(aheadOrigin, -normal * (reach + 0.5), params)

                            if currentWall and math.abs(currentWall.Normal.Y) <= 0.35
                                and (not wallAhead or math.abs(wallAhead.Normal.Y) > 0.35) then
                                cornerInfo = {Normal = normal, Tangent = tangent}
                            end
                        end
                    end
                end
            end

            if cornerInfo then
                if Runtime.CornerArmed
                    and now - Runtime.CornerLastTrigger >= Settings.CornerPop.Cooldown
                    and CanStartBurst("CornerPop", 0.22) then
                    local current = Root.AssemblyLinearVelocity
                    local horizontal = Flat(current)
                    local launchDirection = (cornerInfo.Tangent * 0.72 + cornerInfo.Normal * 0.70)
                    if launchDirection.Magnitude > 0.01 then
                        launchDirection = launchDirection.Unit
                        horizontal += launchDirection * Settings.CornerPop.HorizontalPower
                        CommitVelocity(Vector3.new(
                            horizontal.X,
                            current.Y + Settings.CornerPop.VerticalPower,
                            horizontal.Z
                        ))
                        Runtime.CornerArmed = false
                        Runtime.CornerLastTrigger = now
                        Runtime.CornerRearmAt = now + Settings.CornerPop.Cooldown
                        Runtime.CornerLastPosition = Root.Position
                    else
                        ReleaseBurst("CornerPop")
                    end
                end
            elseif now >= Runtime.CornerRearmAt then
                local lastPos = Runtime.CornerLastPosition
                if not lastPos or (Root.Position - lastPos).Magnitude >= 3 then
                    Runtime.CornerArmed = true
                end
            end
        else
            Runtime.CornerArmed = true
            Runtime.CornerLastPosition = nil
        end

        -- VELOCITY DETONATOR: fire once after crossing the threshold. It
        -- re-arms only after speed falls below the configurable reset band.
        if Settings.VelocityDetonator.Enabled then
            local config = Settings.VelocityDetonator
            local horizontal = Flat(Root.AssemblyLinearVelocity)
            local speed = horizontal.Magnitude
            local threshold = config.Threshold
            if speed < threshold * config.RearmRatio then
                Runtime.DetonatorArmed = true
            end

            if Runtime.DetonatorArmed and speed >= threshold
                and now - Runtime.LastDetonation >= config.Cooldown
                and CanStartBurst("VelocityDetonator", 0.22) then
                local direction = GetDirection(config.DirectionMode)
                if direction.Magnitude > 0.01 then
                    local boosted = horizontal + direction * config.Power
                    local current = Root.AssemblyLinearVelocity
                    CommitVelocity(Vector3.new(boosted.X, current.Y, boosted.Z))
                    Runtime.DetonatorArmed = false
                    Runtime.LastDetonation = now
                else
                    ReleaseBurst("VelocityDetonator")
                end
            end
        else
            Runtime.DetonatorArmed = true
        end

        -- SPIRAL LAUNCH: rotate the horizontal trajectory at a controlled
        -- angular rate while leaving the current vertical physics untouched.
        if Runtime.SpiralActive then
            local config = Settings.SpiralLaunch
            local elapsed = now - Runtime.SpiralStartedAt
            if not config.Enabled or elapsed >= config.Duration then
                Runtime.SpiralActive = false
                ReleaseBurst("SpiralLaunch")
            else
                if Settings.SpeedCapper.Enabled then
                    Runtime.SpiralHorizontalSpeed = math.min(
                        Runtime.SpiralHorizontalSpeed,
                        math.clamp(Settings.SpeedCapper.Cap, Settings.SpeedCapper.Min, Settings.SpeedCapper.Max)
                    )
                end
                local angle = math.rad(config.RotationSpeed * config.Direction * dt)
                local rotated = CFrame.fromAxisAngle(Vector3.yAxis, angle):VectorToWorldSpace(Runtime.SpiralDirection)
                local flat = Flat(rotated)
                if flat.Magnitude > 0.01 then
                    Runtime.SpiralDirection = flat.Unit
                    local current = Root.AssemblyLinearVelocity
                    CommitVelocity(Vector3.new(
                        Runtime.SpiralDirection.X * Runtime.SpiralHorizontalSpeed,
                        current.Y,
                        Runtime.SpiralDirection.Z * Runtime.SpiralHorizontalSpeed
                    ))
                end
            end
        end
    end

    Connect(Player.CharacterAdded, function()
        Runtime.ImpulseCharge = 0
        Runtime.CornerArmed = true
        Runtime.CornerLastTrigger = 0
        Runtime.CornerRearmAt = 0
        Runtime.CornerLastPosition = nil
        Runtime.DetonatorArmed = true
        Runtime.LastDetonation = 0
        Runtime.SpiralActive = false
        Runtime.SpiralLastLaunch = 0
        Runtime.ActiveFeature = nil
        Runtime.BusyUntil = 0
    end)
end

local KeyboardSpammerState = {
    VirtualInput = nil,
    Generation = 0,
    Held = false,
    HeldKey = nil,
    HeldGeneration = nil,
    TriggerHeld = false,
    Status = "OFF",
}

local KeyboardSpammerStatusLabel
local KeyboardSpammerHoldButton
local KeyboardSpammerKeyButton
local syncKeyboardSpammerToggle
local selectingSpammerKey = false

local function UpdateKeyboardSpammerStatus(status)
    if status then
        KeyboardSpammerState.Status = status
    end

    if KeyboardSpammerStatusLabel and KeyboardSpammerStatusLabel.Parent then
        KeyboardSpammerStatusLabel.Text = "Status: " .. tostring(KeyboardSpammerState.Status)
        KeyboardSpammerStatusLabel.TextColor3 = Settings.KeyboardSpammer.Enabled
            and COLORS.AutoTuneOn or COLORS.AutoTuneOff
    end

    if syncKeyboardSpammerToggle then
        syncKeyboardSpammerToggle()
    end
end

local function RefreshKeyboardSpammerKeyButtons()
    if KeyboardSpammerHoldButton and KeyboardSpammerHoldButton.Parent then
        KeyboardSpammerHoldButton.Text = "HOLD: " .. BindingName(Settings.KeyboardSpammer.HoldKey)
    end
    if KeyboardSpammerKeyButton and KeyboardSpammerKeyButton.Parent then
        KeyboardSpammerKeyButton.Text = "SPAM: " .. Settings.KeyboardSpammer.Key.Name
    end
end

local function ReleaseKeyboardSpammerKey(expectedGeneration)
    -- A canceled worker must never release a key pressed by a newer worker.
    if expectedGeneration and KeyboardSpammerState.HeldGeneration ~= expectedGeneration then
        return
    end

    local virtualInput = KeyboardSpammerState.VirtualInput
    local heldKey = KeyboardSpammerState.HeldKey

    -- Clear state first so Stop/re-entrant cleanup cannot leave a stale held flag.
    KeyboardSpammerState.Held = false
    KeyboardSpammerState.HeldKey = nil
    KeyboardSpammerState.HeldGeneration = nil

    if virtualInput and heldKey then
        local ok, err = pcall(function()
            virtualInput:SendKey(false, heldKey, false)
        end)
        if not ok then
            warn("[FortifiedHub] Keyboard Spammer release cleanup failed: " .. tostring(err))
        end
    end
end

StopKeyboardSpammer = function(status)
    Settings.KeyboardSpammer.Enabled = false
    Settings.KeyboardSpammer.ToggleOn = false
    KeyboardSpammerState.TriggerHeld = false
    KeyboardSpammerState.Generation += 1
    ReleaseKeyboardSpammerKey()
    UpdateKeyboardSpammerStatus(status or "OFF")
end

local function GetKeyboardSpammerVirtualInput()
    if KeyboardSpammerState.VirtualInput then
        return KeyboardSpammerState.VirtualInput
    end

    local ok, virtualInputOrError = pcall(function()
        return UIS:CreateVirtualInput()
    end)

    if not ok then
        warn("[FortifiedHub] CreateVirtualInput failed: " .. tostring(virtualInputOrError))
        return nil, "INPUT API ERROR — CHECK OUTPUT"
    end
    if not virtualInputOrError then
        return nil, "INPUT API UNAVAILABLE"
    end

    local hasSendKey = pcall(function()
        return virtualInputOrError.SendKey
    end)
    if not hasSendKey then
        return nil, "SENDKEY UNAVAILABLE"
    end

    KeyboardSpammerState.VirtualInput = virtualInputOrError
    return virtualInputOrError
end

local function StartKeyboardSpammer()
    local manualToggle = Settings.KeyboardSpammer.ToggleOn
    local activationRequested = manualToggle or KeyboardSpammerState.TriggerHeld
    if Settings.KeyboardSpammer.Enabled or not activationRequested then
        return
    end

    local spamKey = Settings.KeyboardSpammer.Key
    local holdBinding = Settings.KeyboardSpammer.HoldKey
    local validSpamKey = typeof(spamKey) == "EnumItem"
        and spamKey.EnumType == Enum.KeyCode
        and spamKey ~= Enum.KeyCode.Unknown
        and spamKey ~= Enum.KeyCode.Escape
    local validHoldBinding = type(holdBinding) == "table"
        and ((holdBinding.Type == Enum.UserInputType.Keyboard
                and typeof(holdBinding.Code) == "EnumItem"
                and holdBinding.Code.EnumType == Enum.KeyCode
                and holdBinding.Code ~= Enum.KeyCode.Unknown
                and holdBinding.Code ~= Enum.KeyCode.Escape)
            or holdBinding.Type == Enum.UserInputType.MouseButton1
            or holdBinding.Type == Enum.UserInputType.MouseButton2
            or holdBinding.Type == Enum.UserInputType.MouseButton3)

    if not validSpamKey or (not manualToggle and not validHoldBinding) then
        Settings.KeyboardSpammer.ToggleOn = false
        UpdateKeyboardSpammerStatus("INVALID KEY")
        return
    end

    if not manualToggle
        and holdBinding.Type == Enum.UserInputType.Keyboard
        and spamKey == holdBinding.Code then
        Settings.KeyboardSpammer.ToggleOn = false
        UpdateKeyboardSpammerStatus("HOLD KEY AND SPAM KEY MUST DIFFER")
        return
    end

    if UIS:GetFocusedTextBox() then
        KeyboardSpammerState.TriggerHeld = false
        Settings.KeyboardSpammer.ToggleOn = false
        UpdateKeyboardSpammerStatus("CLOSE TEXT INPUT FIRST")
        return
    end

    local virtualInput, inputError = GetKeyboardSpammerVirtualInput()
    if not virtualInput then
        Settings.KeyboardSpammer.Enabled = false
        Settings.KeyboardSpammer.ToggleOn = false
        KeyboardSpammerState.TriggerHeld = false
        UpdateKeyboardSpammerStatus(inputError or "INPUT API UNAVAILABLE")
        return
    end

    KeyboardSpammerState.Generation += 1
    local generation = KeyboardSpammerState.Generation
    Settings.KeyboardSpammer.Enabled = true
    UpdateKeyboardSpammerStatus(manualToggle
        and "RUNNING • SWITCH ON"
        or "RUNNING • HOLD " .. BindingName(holdBinding))

    task.spawn(function()
        while Settings.KeyboardSpammer.Enabled
            and KeyboardSpammerState.Generation == generation
            and (Settings.KeyboardSpammer.ToggleOn or KeyboardSpammerState.TriggerHeld) do

            if UIS:GetFocusedTextBox() then
                StopKeyboardSpammer("STOPPED: TEXT INPUT FOCUSED")
                break
            end

            -- For hold-to-run mode, trust our InputBegan/InputEnded flag.
            -- Polling IsKeyDown immediately after InputBegan can report stale
            -- state on some devices and cause a one-press-only failure.
            if not Settings.KeyboardSpammer.ToggleOn and not KeyboardSpammerState.TriggerHeld then
                StopKeyboardSpammer("OFF • HOLD INPUT RELEASED")
                break
            end

            local keyCode = Settings.KeyboardSpammer.Key
            -- Mark the key as needing release before sending. If SendKey throws
            -- after partially dispatching the press, Stop still attempts cleanup.
            KeyboardSpammerState.Held = true
            KeyboardSpammerState.HeldKey = keyCode
            KeyboardSpammerState.HeldGeneration = generation

            local pressOk, pressError = pcall(function()
                virtualInput:SendKey(true, keyCode, false)
            end)
            if not pressOk then
                warn("[FortifiedHub] Keyboard Spammer press failed: " .. tostring(pressError))
                StopKeyboardSpammer("INPUT ERROR — STOPPED")
                break
            end

            local intervalMs = math.clamp(Settings.KeyboardSpammer.IntervalMs, 50, 1000)
            local pressMs = math.clamp(Settings.KeyboardSpammer.PressMs, 20, math.min(100, intervalMs - 5))
            task.wait(pressMs / 1000)

            local stillRequested = Settings.KeyboardSpammer.ToggleOn or KeyboardSpammerState.TriggerHeld
            if not Settings.KeyboardSpammer.Enabled
                or KeyboardSpammerState.Generation ~= generation
                or not stillRequested then
                ReleaseKeyboardSpammerKey(generation)
                if Settings.KeyboardSpammer.Enabled then
                    StopKeyboardSpammer(Settings.KeyboardSpammer.ToggleOn
                        and "STOPPED"
                        or "OFF • HOLD INPUT RELEASED")
                end
                break
            end

            local releaseOk, releaseError = pcall(function()
                virtualInput:SendKey(false, keyCode, false)
            end)
            if not releaseOk then
                warn("[FortifiedHub] Keyboard Spammer release failed: " .. tostring(releaseError))
                StopKeyboardSpammer("RELEASE ERROR — STOPPED")
                break
            end

            if KeyboardSpammerState.HeldGeneration == generation then
                KeyboardSpammerState.Held = false
                KeyboardSpammerState.HeldKey = nil
                KeyboardSpammerState.HeldGeneration = nil
            end

            -- The interval is measured start-to-start, with the key released
            -- for the remainder of the cycle.
            task.wait(math.max(0.01, (intervalMs - pressMs) / 1000))
        end

        -- A final best-effort release covers cancellation while waiting.
        ReleaseKeyboardSpammerKey(generation)
    end)
end

local function BeginSpammerKeyCapture(button, settingsField)
    if selectingSpammerKey then
        return
    end

    if Settings.KeyboardSpammer.Enabled then
        StopKeyboardSpammer("OFF • SELECT KEY")
    end

    selectingSpammerKey = true
    Rebinding = true
    button.Text = "PRESS KEY..."
    UpdateKeyboardSpammerStatus("PRESS A KEY")

    if keyCaptureConnection and keyCaptureConnection.Connected then
        keyCaptureConnection:Disconnect()
    end

    keyCaptureConnection = Connect(UIS.InputBegan, function(input)
        local isKeyboard = input.UserInputType == Enum.UserInputType.Keyboard
        local isMouseButton = input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.MouseButton2
            or input.UserInputType == Enum.UserInputType.MouseButton3

        if settingsField == "Key" and not isKeyboard then
            return
        end
        if settingsField == "HoldKey" and not (isKeyboard or isMouseButton) then
            return
        end

        if isKeyboard and input.KeyCode == Enum.KeyCode.Escape then
            selectingSpammerKey = false
            Rebinding = false
            RefreshKeyboardSpammerKeyButtons()
            if keyCaptureConnection and keyCaptureConnection.Connected then
                keyCaptureConnection:Disconnect()
            end
            UpdateKeyboardSpammerStatus("OFF • SWITCH OFF")
            return
        end

        if isKeyboard and input.KeyCode == Enum.KeyCode.Unknown then
            return
        end

        local newBinding
        if isKeyboard then
            newBinding = {Type = Enum.UserInputType.Keyboard, Code = input.KeyCode}
        else
            newBinding = {Type = input.UserInputType, Code = input.UserInputType}
        end

        local clashes = false
        if settingsField == "HoldKey" then
            clashes = isKeyboard and input.KeyCode == Settings.KeyboardSpammer.Key
        else
            local currentHold = Settings.KeyboardSpammer.HoldKey
            clashes = isKeyboard
                and currentHold.Type == Enum.UserInputType.Keyboard
                and input.KeyCode == currentHold.Code
        end

        if clashes then
            button.Text = "KEYS MUST DIFFER"
            UpdateKeyboardSpammerStatus("HOLD KEY AND SPAM KEY MUST DIFFER")
            return
        end

        if settingsField == "Key" then
            Settings.KeyboardSpammer.Key = input.KeyCode
        else
            Settings.KeyboardSpammer.HoldKey = newBinding
        end

        selectingSpammerKey = false
        Rebinding = false
        if keyCaptureConnection and keyCaptureConnection.Connected then
            keyCaptureConnection:Disconnect()
        end
        RefreshKeyboardSpammerKeyButtons()
        UpdateKeyboardSpammerStatus("OFF • CLICK ON OR HOLD KEY")
    end)
end

--========================================================--
-- FORCE FACING
--========================================================--
-- Keeps the character's horizontal facing aligned with the camera look
-- direction without relying on Roblox's built-in Shift Lock option.
-- It changes yaw only; position and vertical movement are left untouched.
local ForceFacingState = {
    Enabled = false,
    TrackedHumanoid = nil,
    PreviousAutoRotate = nil,
}

local function RestoreForceFacingState()
    ForceFacingState.Enabled = false

    local tracked = ForceFacingState.TrackedHumanoid
    if tracked and tracked.Parent and ForceFacingState.PreviousAutoRotate ~= nil then
        tracked.AutoRotate = ForceFacingState.PreviousAutoRotate
    elseif Humanoid and Humanoid.Parent and ForceFacingState.PreviousAutoRotate ~= nil then
        Humanoid.AutoRotate = ForceFacingState.PreviousAutoRotate
    end

    ForceFacingState.TrackedHumanoid = nil
    ForceFacingState.PreviousAutoRotate = nil
end

RestoreForceFacing = RestoreForceFacingState

local function SetForceFacingEnabled(enabled)
    enabled = enabled == true
    if ForceFacingState.Enabled == enabled then
        return
    end

    if enabled then
        ForceFacingState.Enabled = true
        if Humanoid and Humanoid.Parent then
            ForceFacingState.TrackedHumanoid = Humanoid
            ForceFacingState.PreviousAutoRotate = Humanoid.AutoRotate
            Humanoid.AutoRotate = false
        end
    else
        RestoreForceFacingState()
    end
end

AddCard(utilityGrid, {
    Name = "ForceFacing",
    Title = "🧭  Force Facing",
    Description = "Aligns your character with the camera's forward direction, even when Shift Lock is disabled.",
    Order = 7,
    Height = 232,
    Toggle = true,
    GetEnabled = function()
        return ForceFacingState.Enabled
    end,
    OnToggle = SetForceFacingEnabled,
})

Connect(RunService.RenderStepped, function()
    if not ForceFacingState.Enabled then
        return
    end

    local humanoid = Humanoid
    local root = Root
    if not humanoid or not humanoid.Parent
        or humanoid.Health <= 0
        or not root or not root.Parent then
        return
    end

    -- Rebind after respawn and remember the new humanoid's own AutoRotate value.
    if ForceFacingState.TrackedHumanoid ~= humanoid then
        ForceFacingState.TrackedHumanoid = humanoid
        ForceFacingState.PreviousAutoRotate = humanoid.AutoRotate
    end

    humanoid.AutoRotate = false

    -- Spin Bot owns body rotation while its bind is held; do not fight it.
    if SpinBotState.Holding then
        return
    end

    local camera = workspace.CurrentCamera
    if not camera then
        return
    end

    local look = camera.CFrame.LookVector
    local flatLook = Vector3.new(look.X, 0, look.Z)
    if flatLook.Magnitude < 1e-4 then
        return
    end

    -- Yaw-only alignment: do not copy camera pitch or roll into the character.
    root.CFrame = CFrame.lookAt(
        root.Position,
        root.Position + flatLook.Unit,
        Vector3.yAxis
    )
end)

local function SetActiveTab(name)

    local metadata = TabMetadata[name]
    ContentTitle.Text = name or "FORTIFIEDHUB"
    ContentSubtitle.Text = metadata and metadata.Description or "ABYSS MOVEMENT SYSTEM"

        local selected = tabName == name
        button.BackgroundColor3 = selected and COLORS.TabActive or COLORS.Tab

        local tabLabel = button:FindFirstChild("TabLabel")
        if tabLabel then
            tabLabel.TextColor3 = selected and COLORS.Text or COLORS.Muted
        end

        local indicator = TabIndicators[tabName]
        if indicator then
            indicator.Visible = selected
        end

        local glyph = button:FindFirstChild("TabGlyph")
        if glyph then

                and Color3.fromRGB(17, 79, 94)
                or Color3.fromRGB(13, 46, 61)
            glyph.TextColor3 = selected and Color3.fromRGB(154, 247, 250) or COLORS.Accent
        end
    end

    for pageName, page in pairs(Pages) do
        page.Visible = pageName == name
        if page.Visible then
            page.CanvasPosition = Vector2.zero
        end
    end
end

--========================================================--
-- WALL DETECTION
--========================================================--

FindWall = function(maxCharacterGap)
    if not Root or not Root.Parent then
        return nil
    end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {Character}
    pcall(function()
        params.RespectCanCollide = true
    end)

    local forward = Vector3.new(Root.CFrame.LookVector.X, 0, Root.CFrame.LookVector.Z)
    local right = Vector3.new(Root.CFrame.RightVector.X, 0, Root.CFrame.RightVector.Z)
    if forward.Magnitude < 0.01 then
        forward = Vector3.new(0, 0, -1)
    else
        forward = forward.Unit
    end
    if right.Magnitude < 0.01 then
        right = Vector3.new(1, 0, 0)
    else
        right = right.Unit
    end

    local diagonalA = (forward + right)
    local diagonalB = (forward - right)
    diagonalA = diagonalA.Magnitude > 0.01 and diagonalA.Unit or forward
    diagonalB = diagonalB.Magnitude > 0.01 and diagonalB.Unit or forward

    local directions = {
        forward, -forward, right, -right,
        diagonalA, -diagonalA, diagonalB, -diagonalB,
    }

    -- Sample lower, middle, and upper torso height so slight vertical
    -- offsets or uneven wall geometry do not make Auto Mode miss a wall.
    local verticalOffset = math.max(0.35, Root.Size.Y * 0.35)
    local verticalSamples = {-verticalOffset, 0, verticalOffset}
    local maxGap = maxCharacterGap ~= nil and math.max(0, tonumber(maxCharacterGap) or 0) or nil
    local closest = nil
    local closestMetric = math.huge

    for _, direction in ipairs(directions) do
        local localDirection = Root.CFrame:VectorToObjectSpace(direction)
        local projectedHalfExtent =
            math.abs(localDirection.X) * (Root.Size.X * 0.5)
            + math.abs(localDirection.Z) * (Root.Size.Z * 0.5)

        local rayLength = maxGap ~= nil
            and (projectedHalfExtent + maxGap + 0.12)
            or 5

        for _, yOffset in ipairs(verticalSamples) do
            local origin = Root.Position + Vector3.new(0, yOffset, 0)
            local result = workspace:Raycast(origin, direction * rayLength, params)

            if result and math.abs(result.Normal.Y) <= 0.35 then
                local characterGap = math.max(0, result.Distance - projectedHalfExtent)
                local accepted = maxGap == nil or characterGap <= (maxGap + 0.12)
                local metric = maxGap ~= nil and characterGap or result.Distance

                if accepted and metric < closestMetric then
                    closest = result
                    closestMetric = metric
                end
            end
        end
    end

    return closest
end

--========================================================--
-- RAMP DETECTION
--========================================================--

FindRamp = function()
    if not Root or not Root.Parent then
        return nil
    end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {Character}

    local offsets = {
        Vector3.zero,
        Vector3.new(1.5, 0, 0),
        Vector3.new(-1.5, 0, 0),
        Vector3.new(0, 0, 1.5),
        Vector3.new(0, 0, -1.5),
    }

    local best = nil

    for _, offset in ipairs(offsets) do
        local result = workspace:Raycast(
            Root.Position + offset,
            Vector3.new(0, -7, 0),
            params
        )

        if result then
            local dot = math.clamp(
                result.Normal:Dot(Vector3.yAxis),
                -1,
                1
            )

            local angle = math.deg(math.acos(dot))

            if angle >= 8 and angle <= 70 then
                if not best or result.Distance < best.Distance then
                    best = result
                end
            end
        end
    end

    return best
end

--========================================================--
-- WALL LAUNCH
--========================================================--

AttachWall = function()
    local wall = FindWall()

    if not wall or not Humanoid or not Root then
        return false
    end

    -- Wall Launch takes exclusive ownership from Wall Glide / carried momentum.
    if WallGlideState.Active then
        StopWallGlide()
    end
    WallGlideState.MomentumCarrying = false
    WallGlideState.MomentumCarryUntil = 0

    WallLaunchState.Attached = true
    WallLaunchState.Normal = wall.Normal
    WallLaunchState.Point = wall.Position
    WallLaunchState.LockedY = Root.Position.Y
    WallLaunchState.PreviousSpeed = Humanoid.WalkSpeed

    -- Match Wall Glide's controlled wall movement while attached.
    Humanoid.WalkSpeed = 120

    local velocity = Root.AssemblyLinearVelocity
    local intoWall = wall.Normal * velocity:Dot(wall.Normal)

    Root.AssemblyLinearVelocity = velocity - intoWall

    return true
end

StopWallLaunch = function()
    if Humanoid and WallLaunchState.PreviousSpeed ~= nil then
        Humanoid.WalkSpeed = WallLaunchState.PreviousSpeed
    end

    WallLaunchState.Attached = false
    WallLaunchState.Normal = nil
    WallLaunchState.Point = nil
    WallLaunchState.LockedY = nil
    WallLaunchState.PreviousSpeed = nil
end

LaunchFromWall = function()
    if not WallLaunchState.Attached or not Root then
        return
    end

    local normal = WallLaunchState.Normal
    StopWallLaunch()
    WallGlideState.MomentumCarrying = false
    WallGlideState.MomentumCarryUntil = 0

    local away = Vector3.new(
        normal.X,
        0,
        normal.Z
    )

    if away.Magnitude < 0.01 then
        away = Vector3.new(
            Root.CFrame.LookVector.X,
            0,
            Root.CFrame.LookVector.Z
        )
    end

    if away.Magnitude > 0.01 then
        away = away.Unit
    end

    Root.AssemblyLinearVelocity = Settings._ClampHorizontalVelocity(
        away * Settings.WallLaunch.Power
        + Vector3.new(
            0,
            Settings.WallLaunch.Power * 0.65,
            0
        )
    )
end

--========================================================--
-- WALL GLIDE
--========================================================--

StartWallGlide = function(wall)
    wall = wall or FindWall()

    if not wall or not Humanoid or not Root then
        return false
    end

    if WallLaunchState.Attached then
        StopWallLaunch()
    end
    WallGlideState.MomentumCarrying = false
    WallGlideState.MomentumCarryUntil = 0

    WallGlideState.Active = true
    WallGlideState.Normal = wall.Normal
    WallGlideState.Point = wall.Position
    WallGlideState.PreviousSpeed = Humanoid.WalkSpeed
    WallGlideState.LockedY = Root.Position.Y

    local move = Humanoid.MoveDirection
    local velocity = Root.AssemblyLinearVelocity
    local horizontalVelocity = Vector3.new(velocity.X, 0, velocity.Z)
    local tangent = move - wall.Normal * move:Dot(wall.Normal)

    if tangent.Magnitude < 0.05 then
        tangent = horizontalVelocity - wall.Normal * horizontalVelocity:Dot(wall.Normal)
    end
    if tangent.Magnitude < 0.05 then
        local look = Root.CFrame.LookVector
        tangent = look - wall.Normal * look:Dot(wall.Normal)
    end
    if tangent.Magnitude < 0.05 then
        local right = Root.CFrame.RightVector
        tangent = right - wall.Normal * right:Dot(wall.Normal)
    end

    WallGlideState.TangentDirection = tangent.Magnitude > 0.05 and tangent.Unit or nil
    WallGlideState.MomentumCarrying = false
    WallGlideState.MomentumCarryUntil = 0
    if WallGlideState.TangentDirection then
        local alongSpeed = math.max(0, horizontalVelocity:Dot(WallGlideState.TangentDirection))
        WallGlideState.MomentumSpeed = math.max(70, alongSpeed)
        WallGlideState.MomentumDirection = WallGlideState.TangentDirection
    else
        WallGlideState.MomentumSpeed = 70
        WallGlideState.MomentumDirection = nil
    end
    Humanoid.WalkSpeed = 120

    return true
end

StopWallGlide = function()
    if Humanoid and WallGlideState.PreviousSpeed ~= nil then
        Humanoid.WalkSpeed = WallGlideState.PreviousSpeed
    end

    -- Capture the final wall-run speed before clearing the attachment data.
    -- The carry loop preserves it and adds speed while the player continues
    -- steering generally along the last valid wall direction.
    if Settings.WallMomentum.Enabled and Root and Root.Parent then
        local currentVelocity = Root.AssemblyLinearVelocity
        local horizontal = Vector3.new(currentVelocity.X, 0, currentVelocity.Z)
        local direction = WallGlideState.TangentDirection or WallGlideState.MomentumDirection

        if direction and direction.Magnitude > 0.05 then
            direction = direction.Unit
            local alongSpeed = math.max(0, horizontal:Dot(direction))
            local carrySpeed = math.max(WallGlideState.MomentumSpeed or 70, alongSpeed)
            local maxSpeed = Settings.WallMomentum.MaxSpeed
            if Settings.SpeedCapper.Enabled then
                maxSpeed = math.min(maxSpeed, Settings.SpeedCapper.Cap)
            end

            WallGlideState.MomentumSpeed = math.clamp(carrySpeed, 0, maxSpeed)
            WallGlideState.MomentumDirection = direction
            WallGlideState.MomentumCarrying = true
            WallGlideState.MomentumCarryUntil = os.clock() + Settings.WallMomentum.CarryDuration
        end
    end

    WallGlideState.Active = false
    WallGlideState.Normal = nil
    WallGlideState.Point = nil
    WallGlideState.LockedY = nil
    WallGlideState.PreviousSpeed = nil
    WallGlideState.TangentDirection = nil
end

-- A launch takes control away from wall controllers so their next Heartbeat
-- cannot overwrite the new propulsion impulse with a stale glide/carry vector.
Settings._PrepareForPropulsion = function()
    if WallGlideState.Active then
        StopWallGlide()
    end
    if WallLaunchState.Attached and StopWallLaunch then
        StopWallLaunch()
    end
    WallGlideState.MomentumCarrying = false
    WallGlideState.MomentumCarryUntil = 0
end

--========================================================--
-- RAMP LAUNCH
--========================================================--

DoRampLaunch = function(ramp)
    if not Root or not Root.Parent or not Humanoid then
        return
    end

    -- HARD RELEASE-ONLY GATE:
    -- the launch cannot happen while the configured key is held, and it
    -- cannot happen unless the hold phase actually armed on a ramp.
    if not Settings.RampLaunch.Enabled
        or RampLaunchState.Launching
        or RampLaunchState.Holding
        or not RampLaunchState.ReleaseRequested
        or not RampLaunchState.Armed then
        return
    end

    if os.clock() - RampLaunchState.LastLaunch < 0.35 then
        return
    end

    ramp = ramp or RampLaunchState.DetectedRamp
    if not ramp then
        return
    end

    if Settings._PrepareForPropulsion then
        Settings._PrepareForPropulsion()
    end

    RampLaunchState.Launching = true
    RampLaunchState.LastLaunch = os.clock()

    -- The character has already been facing the camera's horizontal
    -- direction during the hold phase. Preserve that facing for launch.
    local direction = Vector3.new(
        Root.CFrame.LookVector.X,
        0,
        Root.CFrame.LookVector.Z
    )

    if direction.Magnitude <= 0.01 then
        RampLaunchState.Launching = false
        return
    end

    direction = direction.Unit

    local power = math.clamp(
        Settings.RampLaunch.Power,
        Settings.RampLaunch.Min,
        Settings.RampLaunch.Max
    )

    local forwardPower = math.clamp(
        Settings.RampLaunch.ForwardPower,
        Settings.RampLaunch.ForwardMin,
        Settings.RampLaunch.ForwardMax
    )

    Root.AssemblyAngularVelocity = Vector3.zero
    Root.AssemblyLinearVelocity = Settings._ClampHorizontalVelocity(
        direction * forwardPower
        + Vector3.new(0, power, 0)
    )

    RampLaunchState.DetectedRamp = nil
    RampLaunchState.Armed = false
    RampLaunchState.ReleaseRequested = false
    RampLaunchState.Launching = false
end

--========================================================--
-- RAMP BOUNCE
--========================================================--

DoRampBounce = function(ramp)
    if not Root or not Root.Parent then
        return
    end

    if Settings._PrepareForPropulsion then
        Settings._PrepareForPropulsion()
    end

    -- Restore the older Ramp Bounce behavior:
    -- keep the current horizontal velocity, then add the bounce impulse.
    -- The forward boost follows the ramp's horizontal normal.
    local current = Root.AssemblyLinearVelocity
    local horizontal = Vector3.new(current.X, 0, current.Z)

    local normal = ramp.Normal
    local rampDirection = Vector3.new(normal.X, 0, normal.Z)

    if rampDirection.Magnitude > 0.05 then
        rampDirection = rampDirection.Unit
    end

    local velocity =
        horizontal
        + Vector3.new(0, Settings.RampBounce.Power, 0)

    local forwardPower = math.clamp(
        Settings.RampBounce.ForwardPower,
        Settings.RampBounce.ForwardMin,
        Settings.RampBounce.ForwardMax
    )

    if rampDirection.Magnitude > 0.05 then
        velocity += rampDirection * forwardPower
    end

    Root.AssemblyLinearVelocity = Settings._ClampHorizontalVelocity(velocity)
end


-- RAPTURE BOUNCE
-- A separate, one-shot high-power vertical launch. Unlike Ramp Bounce,
-- it does not repeatedly fire every Heartbeat while the key remains held.
DoRaptureBounce = function(ramp, instant)
    if not Settings.RaptureBounce.Enabled
        or (not instant and (not RaptureBounceState.Holding or RaptureBounceState.TriggeredForHold))
        or not ramp
        or not Root
        or not Root.Parent
        or not Humanoid
        or Humanoid.Health <= 0 then
        return false
    end

    local now = os.clock()
    local cooldown = math.clamp(
        Settings.RaptureBounce.Cooldown,
        Settings.RaptureBounce.CooldownMin,
        Settings.RaptureBounce.CooldownMax
    )

    if RaptureBounceState.LastLaunch > 0
        and now - RaptureBounceState.LastLaunch < cooldown then
        return false
    end

    if Settings._PrepareForPropulsion then
        Settings._PrepareForPropulsion()
    end

    local current = Root.AssemblyLinearVelocity
    local targetHeight = math.clamp(
        Settings.RaptureBounce.TargetHeight,
        Settings.RaptureBounce.TargetHeightMin,
        Settings.RaptureBounce.TargetHeightMax
    )

    local forwardPower = math.clamp(
        Settings.RaptureBounce.ForwardPower,
        Settings.RaptureBounce.ForwardPowerMin,
        Settings.RaptureBounce.ForwardPowerMax
    )

    -- Set the launch velocity from the selected target rise, not as an
    -- additive height bonus. In ideal Roblox gravity, v = sqrt(2 * g * h),
    -- so the selected value represents the approximate apex rise in studs
    -- above the character's launch position.
    local gravity = math.max(workspace.Gravity, 0)
    if gravity <= 0.001 then
        return false
    end
    local targetVerticalSpeed = math.sqrt(2 * gravity * targetHeight)

    -- Add forward power along the character's horizontal facing direction
    -- while retaining existing horizontal momentum. Vertical speed is set
    -- from TargetHeight so this control directly determines the launch apex.
    local forwardDirection = Root.CFrame.LookVector
    forwardDirection = Vector3.new(forwardDirection.X, 0, forwardDirection.Z)

    if forwardDirection.Magnitude > 0.001 then
        forwardDirection = forwardDirection.Unit
    else
        forwardDirection = Vector3.zero
    end

    local horizontal = Vector3.new(current.X, 0, current.Z)
        + forwardDirection * forwardPower

    Root.AssemblyLinearVelocity = Settings._ClampHorizontalVelocity(Vector3.new(
        horizontal.X,
        targetVerticalSpeed,
        horizontal.Z
    ))
    Root.AssemblyAngularVelocity = Vector3.zero

    RaptureBounceState.LastLaunch = now
    if not instant or RaptureBounceState.Holding then
        RaptureBounceState.TriggeredForHold = true
    end
    return true
end

Connect(UIS.InputBegan, function(input, processed)
    if Rebinding then
        return
    end

    -- Capture Ramp Launch even if another Roblox action has marked the input
    -- as processed. This prevents the release-only latch from missing a press.
    if Settings.RampLaunch.Enabled
        and InputMatches(Settings.RampLaunch.Key, input) then

        RampLaunchState.Holding = true
        RampLaunchState.DetectedRamp = nil
        RampLaunchState.Armed = false
        RampLaunchState.ReleaseRequested = false
        return
    end

    -- Capture Rapture Bounce's bind before the processed-input gate so it
    -- remains reliable if the game also consumes the selected key.
    if Settings.RaptureBounce.Enabled
        and InputMatches(Settings.RaptureBounce.Key, input) then
        RaptureBounceState.Holding = true
        RaptureBounceState.TriggeredForHold = false
        return
    end

    -- Wall Glide must receive its key even when the experience marks that
    -- input as processed. In Auto Mode the key only performs keybind-un-glide;
    -- in Manual Mode it toggles the glide as before.
    if Settings.WallGlide.Enabled
        and InputMatches(Settings.WallGlide.Key, input) then

        if Settings.WallGlide.AutoMode then
            if Settings.WallGlide.UnglideMode == "Keybind" and WallGlideState.Active then
                StopWallGlide()
            end
        elseif WallGlideState.Active then
            StopWallGlide()
        else
            StartWallGlide()
        end
    end

    if processed then
        return
    end

    if Settings.WallLaunch.Enabled
        and InputMatches(Settings.WallLaunch.Key, input) then

        if WallLaunchState.Attached then
            LaunchFromWall()
        else
            AttachWall()
        end
    end

    if Settings.RampBounce.Enabled
        and InputMatches(Settings.RampBounce.Key, input) then

        RampBounceState.Holding = true
    end

    if Settings.AirStrafe.Enabled
        and InputMatches(Settings.AirStrafe.Key, input) then

        AirStrafeState.Holding = true
        Settings.AirStrafe.Holding = true
    end

    -- SpinBot input is polled every frame in the SpinBot loop below.
    -- Do not react to ordinary W/A/S/D input here.
end)

Connect(UIS.InputEnded, function(input)
    if InputMatches(Settings.RampLaunch.Key, input) then
        -- InputEnded only marks the release. The launch happens in the
        -- release-only loop after the bind is confirmed up.
        RampLaunchState.Holding = false
        RampLaunchState.ReleaseRequested = true
    end

    if InputMatches(Settings.RampBounce.Key, input) then
        RampBounceState.Holding = false
    end

    if InputMatches(Settings.RaptureBounce.Key, input) then
        RaptureBounceState.Holding = false
        RaptureBounceState.TriggeredForHold = false
    end

    if InputMatches(Settings.AirStrafe.Key, input) then
        AirStrafeState.Holding = false
        Settings.AirStrafe.Holding = false
    end

    -- SpinBot holding state is polled every frame, so an InputEnded event
    -- is not required to stop the feature. This also prevents W/S from
    -- affecting SpinBot.

end)

--========================================================--
-- RAMP LAUNCH HOLD / FACE / RELEASE
--========================================================--

-- Hold phase:
--   * detect a ramp
--   * face the camera's horizontal forward direction
--   * cancel velocity through the ramp normal so the character stays pressed
--     to the ramp
--   * NEVER launch while the key is held
--
-- Release phase:
--   * launch once, using the facing direction established during the hold
--
local function FaceRampForward()
    if not Root or not Root.Parent then
        return
    end

    local camera = workspace.CurrentCamera
    local look = camera and camera.CFrame.LookVector or Root.CFrame.LookVector
    local forward = Vector3.new(look.X, 0, look.Z)

    if forward.Magnitude > 0.01 then
        forward = forward.Unit
        Root.CFrame = CFrame.lookAt(
            Root.Position,
            Root.Position + forward
        )
    end
end

Connect(RunService.Heartbeat, function()
    if not Settings.RampLaunch.Enabled then
        RampLaunchState.Holding = false
        RampLaunchState.ReleaseRequested = false
        RampLaunchState.DetectedRamp = nil
        RampLaunchState.Armed = false
        return
    end

    if not Root
        or not Root.Parent
        or not Humanoid
        or Humanoid.Health <= 0 then
        return
    end

    -- HOLD: this branch contains no launch call by design.
    if RampLaunchState.Holding then
        local ramp = FindRamp()
        if ramp then
            RampLaunchState.DetectedRamp = ramp
            RampLaunchState.Armed = true
            FaceRampForward()

            local normal = ramp.Normal
            if normal.Magnitude > 0.01 then
                normal = normal.Unit
                local velocity = Root.AssemblyLinearVelocity
                local normalVelocity = velocity:Dot(normal)

                -- Remove the component of velocity through the ramp while
                -- preserving movement along the ramp surface.
                Root.AssemblyLinearVelocity =
                    velocity - normal * normalVelocity
            end
        end

        return
    end

    -- RELEASE: only this branch can launch Ramp Launch.
    if RampLaunchState.ReleaseRequested
        and RampLaunchState.Armed then

        local ramp = FindRamp() or RampLaunchState.DetectedRamp
        if ramp then
            DoRampLaunch(ramp)
        else
            RampLaunchState.ReleaseRequested = false
            RampLaunchState.Armed = false
            RampLaunchState.DetectedRamp = nil
        end
    end
end)

--========================================================--
-- AUTO JUMP INPUT
--========================================================--

Connect(UIS.InputBegan, function(input)
    if Rebinding then
        return
    end

    if input.UserInputType == Enum.UserInputType.Keyboard
        and input.KeyCode == Enum.KeyCode.Space then

        Settings.AutoJump.SpaceHeld = true
        AutoJumpState.CanJump = true
        AutoJumpState.WasAirborne = false
    end
end)

        and input.KeyCode == Enum.KeyCode.Space then

        Settings.AutoJump.SpaceHeld = false
        AutoJumpState.CanJump = true
        AutoJumpState.WasAirborne = false
    end
end)

-- WALL LAUNCH LOOP
--========================================================--

Connect(RunService.Heartbeat, function()
    if not WallLaunchState.Attached then
        return
    end

    if not Settings.WallLaunch.Enabled
        or not Root
        or not Root.Parent
        or not Humanoid
        or Humanoid.Health <= 0 then
        StopWallLaunch()
        return
    end

    local normal = WallLaunchState.Normal
    local point = WallLaunchState.Point

    if not normal or not point then
        StopWallLaunch()
        return
    end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {Character}
    pcall(function() params.RespectCanCollide = true end)
    local wallCheck = workspace:Raycast(Root.Position, -normal * 5, params)
    if not wallCheck
        or math.abs(wallCheck.Normal.Y) > 0.35
        or wallCheck.Normal:Dot(normal) < 0.55 then
        StopWallLaunch()
        return
    end

    -- Use the same controlled tangent movement as Wall Glide.
    local move = Humanoid.MoveDirection
    local tangent = move - normal * move:Dot(normal)

    local velocity = Vector3.zero

    if tangent.Magnitude > 0.05 then
        tangent = tangent.Unit
        velocity = tangent * 70
    end

    Root.AssemblyLinearVelocity = Vector3.new(
        velocity.X,
        0,
        velocity.Z
    )

    -- Keep the same vertical level while attached.
    if WallLaunchState.LockedY then
        local position = Root.Position

        Root.CFrame =
            CFrame.new(
                position.X,
                WallLaunchState.LockedY,
                position.Z
            ) *
            Root.CFrame.Rotation
    end

    -- Maintain a stable distance from the wall instead of repeatedly
    -- cancelling and rebuilding the whole velocity vector.
    local distance =
        (Root.Position - point):Dot(normal)

    local correction = 2.2 - distance

    if math.abs(correction) > 0.02 then
        Root.AssemblyLinearVelocity +=
            normal * correction * 18
    end
end)

--========================================================--
-- WALL GLIDE LOOP
--========================================================--

Connect(RunService.Heartbeat, function(deltaTime)
    if WallLaunchState.Attached then
        if WallGlideState.Active then
            StopWallGlide()
        end
        return
    end

    if not Settings.WallGlide.Enabled then
        if WallGlideState.Active then
            StopWallGlide()
        end
        return
    end

    if not Root or not Root.Parent or not Humanoid or Humanoid.Health <= 0 then
        StopWallGlide()
        return
    end

    -- Auto Mode is a proximity-based attach: the character must already be
    -- within the configured clearance. It never teleports or pulls toward a
    -- distant wall. Toggling Auto Mode also turns on the master switch.
    if Settings.WallGlide.AutoMode and not WallGlideState.Active then
        local candidate = FindWall(Settings.WallGlide.DetectionDistance)
        if candidate then
            StartWallGlide(candidate)
        end
        return
    end

    if not WallGlideState.Active then
        return
    end

    local normal = WallGlideState.Normal
    local point = WallGlideState.Point
    if not normal or not point then
        StopWallGlide()
        return
    end

    local move = Humanoid.MoveDirection
    local tangent = move - normal * move:Dot(normal)
    local currentVelocity = Root.AssemblyLinearVelocity
    local horizontalVelocity = Vector3.new(currentVelocity.X, 0, currentVelocity.Z)

    if tangent.Magnitude > 0.05 then
        WallGlideState.TangentDirection = tangent.Unit
    elseif Settings.WallGlide.AutoMode then
        local velocityTangent = horizontalVelocity - normal * horizontalVelocity:Dot(normal)
        if velocityTangent.Magnitude > 0.5 then
            WallGlideState.TangentDirection = velocityTangent.Unit
        end

        if WallGlideState.TangentDirection then
            tangent = WallGlideState.TangentDirection
        else
            local look = Root.CFrame.LookVector
            tangent = look - normal * look:Dot(normal)
            if tangent.Magnitude < 0.05 then
                local right = Root.CFrame.RightVector
                tangent = right - normal * right:Dot(normal)
            end
            if tangent.Magnitude > 0.05 then
                tangent = tangent.Unit
                WallGlideState.TangentDirection = tangent
            end
        end
    end

    -- Auto Unglide stops at the end of a wall. Probe at the character and
    -- slightly ahead along the last valid tangent so standing still or
    -- pressing directly into the wall does not produce a zero-length probe.
    if Settings.WallGlide.AutoMode and Settings.WallGlide.UnglideMode == "Auto" then
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = {Character}
        pcall(function()
            params.RespectCanCollide = true
        end)

        local wallReach = 6
        local currentWall = workspace:Raycast(Root.Position, -normal * wallReach, params)
        if not currentWall or math.abs(currentWall.Normal.Y) > 0.35 then
            StopWallGlide()
            return
        end

        local probeDirection = tangent
        if probeDirection.Magnitude <= 0.05 and WallGlideState.TangentDirection then
            probeDirection = WallGlideState.TangentDirection
        end
        if probeDirection.Magnitude > 0.05 then
            local aheadOrigin = Root.Position + probeDirection.Unit * 1.5
            local wallAhead = workspace:Raycast(aheadOrigin, -normal * wallReach, params)
            if not wallAhead or math.abs(wallAhead.Normal.Y) > 0.35 then
                StopWallGlide()
                return
            end
        end

        -- Follow gradual wall curves, but let Auto Unglide handle sharp ends
        -- and corners instead of snapping the character around them.
        if currentWall.Normal:Dot(normal) > 0.65 then
            WallGlideState.Normal = currentWall.Normal
            WallGlideState.Point = currentWall.Position
            normal = currentWall.Normal
            point = currentWall.Position
        end
    end

    local velocity = Vector3.zero
    if tangent.Magnitude > 0.05 then
        tangent = tangent.Unit
        local glideSpeed = 70

        if Settings.WallMomentum.Enabled then
            local dt = math.clamp(deltaTime or (1 / 60), 0, 0.1)
            local moveMagnitude = move.Magnitude
            if moveMagnitude > 0.05 then
                local moveUnit = move / moveMagnitude
                local wallNormalInput = moveUnit:Dot(normal)
                local alongWallInput = (moveUnit - normal * wallNormalInput).Magnitude
                -- A diagonal that presses into the wall and runs along it
                -- receives the strongest acceleration (roughly a 45-degree approach).
                local intoWall = math.max(0, -wallNormalInput)
                local angleQuality = math.clamp(alongWallInput * intoWall * 2, 0, 1)
                if angleQuality > 0.02 then
                    WallGlideState.MomentumSpeed +=
                        Settings.WallMomentum.WallAcceleration * angleQuality * dt
                end
            end

            local alongCurrentSpeed = math.max(0, horizontalVelocity:Dot(tangent))
            WallGlideState.MomentumSpeed = math.max(
                WallGlideState.MomentumSpeed or 70,
                alongCurrentSpeed,
                70
            )

            local maxSpeed = Settings.WallMomentum.MaxSpeed
            if Settings.SpeedCapper.Enabled then
                maxSpeed = math.min(maxSpeed, Settings.SpeedCapper.Cap)
            end
            glideSpeed = math.clamp(WallGlideState.MomentumSpeed, 0, maxSpeed)
            WallGlideState.MomentumSpeed = glideSpeed
            WallGlideState.MomentumDirection = tangent
        else
            WallGlideState.MomentumSpeed = 70
            WallGlideState.MomentumDirection = tangent
        end

        velocity = tangent * glideSpeed
    elseif Settings.WallGlide.AutoMode and WallGlideState.TangentDirection then
        local fallbackSpeed = Settings.WallMomentum.Enabled
            and math.max(70, WallGlideState.MomentumSpeed or 70) or 70
        velocity = WallGlideState.TangentDirection * fallbackSpeed
        if Settings.WallMomentum.Enabled then
            WallGlideState.MomentumDirection = WallGlideState.TangentDirection
        end
    end

    Root.AssemblyLinearVelocity = Vector3.new(velocity.X, 0, velocity.Z)

    local position = Root.Position
    Root.CFrame = CFrame.new(position.X, WallGlideState.LockedY, position.Z) * Root.CFrame.Rotation

    local targetDistance = 2.2
    if Settings.WallGlide.AutoMode then
        local localNormal = Root.CFrame:VectorToObjectSpace(normal)
        local bodyHalfExtent =
            math.abs(localNormal.X) * (Root.Size.X * 0.5)
            + math.abs(localNormal.Z) * (Root.Size.Z * 0.5)
        targetDistance = bodyHalfExtent + Settings.WallGlide.DetectionDistance
    end

    local distance = (Root.Position - point):Dot(normal)
    local correction = targetDistance - distance
    if math.abs(correction) > 0.02 then
        local correctionSpeed = Settings.WallGlide.AutoMode and 14 or 20
        Root.AssemblyLinearVelocity += normal * math.clamp(correction * correctionSpeed, -30, 30)
    end
end)

--========================================================--
-- WALL MOMENTUM CARRY LOOP
--========================================================--
-- Keeps speed earned during a diagonal wall run after detaching. Holding a
-- generally aligned movement direction adds acceleration during the carry.
Connect(RunService.Heartbeat, function(deltaTime)
    if not Settings.WallMomentum.Enabled then
        WallGlideState.MomentumCarrying = false
        return
    end

    if not Root or not Root.Parent or not Humanoid or Humanoid.Health <= 0 then
        WallGlideState.MomentumCarrying = false
        WallGlideState.MomentumSpeed = 70
        WallGlideState.MomentumDirection = nil
        return
    end

    if WallGlideState.Active or not WallGlideState.MomentumCarrying then
        return
    end

    if os.clock() >= (WallGlideState.MomentumCarryUntil or 0) then
        WallGlideState.MomentumCarrying = false
        WallGlideState.MomentumSpeed = 70
        WallGlideState.MomentumDirection = nil
        return
    end

    -- Do not compete with an explicit wall attach or a ramp launch controller.
    if WallLaunchState.Attached or RampLaunchState.Launching then
        return
    end

    local direction = WallGlideState.MomentumDirection
    if not direction or direction.Magnitude <= 0.05 then
        WallGlideState.MomentumCarrying = false
        return
    end
    direction = Vector3.new(direction.X, 0, direction.Z)
    if direction.Magnitude <= 0.05 then
        WallGlideState.MomentumCarrying = false
        return
    end
    direction = direction.Unit

    local velocity = Root.AssemblyLinearVelocity
    local horizontal = Vector3.new(velocity.X, 0, velocity.Z)
    local move = Vector3.new(Humanoid.MoveDirection.X, 0, Humanoid.MoveDirection.Z)
    local dt = math.clamp(deltaTime or (1 / 60), 0, 0.1)

    if move.Magnitude > 0.05 then
        local alignment = move.Unit:Dot(direction)
        -- A deliberate reverse input returns control to normal movement.
        if alignment < -0.35 then
            WallGlideState.MomentumCarrying = false
            WallGlideState.MomentumSpeed = 70
            WallGlideState.MomentumDirection = nil
            return
        end

        if alignment > 0.15 then
            local steerAlpha = math.clamp(dt * 2.5, 0, 0.2)
            local steered = direction:Lerp(move.Unit, steerAlpha)
            if steered.Magnitude > 0.05 then
                direction = steered.Unit
            end
            WallGlideState.MomentumSpeed +=
                Settings.WallMomentum.CarryAcceleration * math.clamp(alignment, 0, 1) * dt
        end
    end

    -- Adopt only velocity that is traveling along the carried direction.
    -- This avoids turning unrelated sideways/forward boosts into a fake speed gain.
    local currentAlongSpeed = math.max(0, horizontal:Dot(direction))
    if currentAlongSpeed > WallGlideState.MomentumSpeed then
        WallGlideState.MomentumSpeed = currentAlongSpeed
    end

    local maxSpeed = Settings.WallMomentum.MaxSpeed
    if Settings.SpeedCapper.Enabled then
        maxSpeed = math.min(maxSpeed, Settings.SpeedCapper.Cap)
    end
    local targetSpeed = math.clamp(
        math.max(WallGlideState.MomentumSpeed, currentAlongSpeed),
        0,
        maxSpeed
    )

    -- Avoid forcing momentum through a solid obstruction directly ahead.
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {Character}
    local obstacle = workspace:Raycast(Root.Position, direction * 2.25, params)
    if obstacle and obstacle.Normal.Y < 0.35 and direction:Dot(obstacle.Normal) < -0.2 then
        WallGlideState.MomentumCarrying = false
        WallGlideState.MomentumSpeed = 70
        WallGlideState.MomentumDirection = nil
        return
    end

    WallGlideState.MomentumSpeed = targetSpeed
    WallGlideState.MomentumDirection = direction
    Root.AssemblyLinearVelocity = Vector3.new(
        direction.X * targetSpeed,
        velocity.Y,
        direction.Z * targetSpeed
    )
end)

--========================================================--
-- GROUND CHECK
--========================================================--

IsCharacterGrounded = function()
    if not Root
        or not Root.Parent
        or not Humanoid then

        return false
    end

    -- Trust Roblox's own floor state first.
    if Humanoid.FloorMaterial ~= Enum.Material.Air then
        return true
    end

    -- During Evade-style sliding, do NOT use the fallback raycast.
    -- The slide controller can move a collider/object upward underneath
    -- the character, which makes the old raycast report a false landing.
    -- That false landing can cause AutoJump to request repeated jumps.
    local sliding =
        UIS:IsKeyDown(Enum.KeyCode.LeftControl)
        or UIS:IsKeyDown(Enum.KeyCode.RightControl)
        or UIS:IsKeyDown(Enum.KeyCode.C)

    if sliding then
        return false
    end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {Character}

    local distance =
        math.max(
            Humanoid.HipHeight + 2.5,
            4.5
        )

    return workspace:Raycast(
        Root.Position,
        Vector3.new(0, -distance, 0),
        params
    ) ~= nil
end

--========================================================--
-- AUTO JUMP
--========================================================--
-- Uses the game's normal Humanoid jump request.
-- Slide-aware: Ctrl/C are treated as slide-held states so the
-- auto-jump request is allowed while the player is sliding.
-- No vertical velocity is forced by AutoJump.

local AutoJumpSlideState = {
    LastRequest = 0,
    WasGrounded = false,
}

local function IsSlideKeyHeld()
    return UIS:IsKeyDown(Enum.KeyCode.LeftControl)
        or UIS:IsKeyDown(Enum.KeyCode.RightControl)
        or UIS:IsKeyDown(Enum.KeyCode.C)
end

local function RequestNormalJump()
    if not Humanoid
        or Humanoid.Health <= 0
        or not Root
        or not Root.Parent then
        return
    end

    Humanoid.Jump = true
    Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
end

PerformAutoJump = function()
    RequestNormalJump()
end

Connect(RunService.Heartbeat, function()
    if not Settings.AutoJump.Enabled then
        AutoJumpState.CanJump = true
        AutoJumpState.WasAirborne = false
        AutoJumpSlideState.WasGrounded = false
        Settings.AutoJump.SpaceHeld = false
        return
    end

    if not Humanoid
        or Humanoid.Health <= 0
        or not Root
        or not Root.Parent then
        return
    end

    local spaceHeld = UIS:IsKeyDown(Enum.KeyCode.Space)
    Settings.AutoJump.SpaceHeld = spaceHeld

    if not spaceHeld then
        AutoJumpState.CanJump = true
        AutoJumpState.WasAirborne = false
        AutoJumpSlideState.WasGrounded = false
        return
    end

    local grounded = IsCharacterGrounded()
    local sliding = IsSlideKeyHeld()
    local now = os.clock()

    -- Normal bhop path: preserve the working landing-based behavior.
    if grounded and not sliding then
        if not AutoJumpState.WasAirborne then
            if AutoJumpState.CanJump then
                AutoJumpState.CanJump = false
                PerformAutoJump()
            end
        else
            AutoJumpState.WasAirborne = false
            AutoJumpState.CanJump = true
            PerformAutoJump()
        end
    elseif not grounded then
        AutoJumpState.WasAirborne = true
        AutoJumpState.CanJump = false
    end

    -- Slide path: Ctrl/C can keep the player in a slide state where the
    -- normal grounded transition isn't reliable. Issue a normal jump
    -- request at a small debounce, but only while actually grounded.
    if sliding and grounded then
        if now - AutoJumpSlideState.LastRequest >= 0.10 then
            AutoJumpSlideState.LastRequest = now
            AutoJumpSlideState.WasGrounded = true
            RequestNormalJump()
        end
    elseif not grounded then
        AutoJumpSlideState.WasGrounded = false
    end
end)

-- SPIN BOT LOOP
--========================================================--
-- SpinBot continuously rotates ONLY while the configured SpinBot key is
-- physically held. The state is polled so processed input cannot break it.
-- W/S are never treated as SpinBot input.

Connect(RunService.RenderStepped, function(dt)
    if not Settings.SpinBot.Enabled then
        SpinBotState.Holding = false

        if Humanoid and SpinBotState.PreviousAutoRotate ~= nil then
            Humanoid.AutoRotate = SpinBotState.PreviousAutoRotate
            SpinBotState.PreviousAutoRotate = nil
        end
        return
    end

    SpinBotState.Holding = IsSpinBindingHeld(Settings.SpinBot.Key)

    if not SpinBotState.Holding then
        if Humanoid and SpinBotState.PreviousAutoRotate ~= nil then
            Humanoid.AutoRotate = SpinBotState.PreviousAutoRotate
            SpinBotState.PreviousAutoRotate = nil
        end
        return
    end

    if not Character
        or not Character.Parent
        or not Root
        or not Root.Parent
        or not Humanoid
        or Humanoid.Health <= 0 then
        return
    end

    local speed = math.clamp(
        Settings.SpinBot.Speed,
        Settings.SpinBot.SpeedMin,
        Settings.SpinBot.SpeedMax
    )

    if speed <= 0 then
        return
    end

    if SpinBotState.PreviousAutoRotate == nil then
        SpinBotState.PreviousAutoRotate = Humanoid.AutoRotate
    end
    Humanoid.AutoRotate = false

    local direction = Settings.SpinBot.Direction < 0 and -1 or 1
    local angleStep = math.rad(speed) * dt * direction

    -- Pivot the whole character model so the body visibly spins instead of
    -- relying on HumanoidRootPart CFrame being the final transform each frame.
    Character:PivotTo(Character:GetPivot() * CFrame.Angles(0, angleStep, 0))
end)

Connect(RunService.Heartbeat, function(dt)

        or not Root.Parent
        or not Humanoid
        or Humanoid.Health <= 0 then

    end

    end

    end

        Root.AssemblyLinearVelocity

            50

            Settings.AirStrafe.Strength,

            (strength / 100) * 10 * dt,

        horizontal:Lerp(
            desired,
            alpha
        )

    local lift =
        math.clamp(
            Settings.AirStrafe.Lift,
            0,
            100
        )

    local newY =
        current.Y + (lift * dt)

    Root.AssemblyLinearVelocity =
        Vector3.new(
            steered.X,
            newY,
            steered.Z
        )
end)

--========================================================--
-- STRAFE OPTIMIZER AUTOTUNE LOOP
--========================================================--
-- Collect a stable sample of camera turn rates without changing sensitivity
-- during the measurement window. At the end, use the median (to reject flick
-- outliers) and apply a single conservative multiplier. This avoids the old
-- feedback loop where each adjustment altered the very rate being measured.

Connect(RunService.RenderStepped, function(dt)
    if not Settings.StrafeOptimizer.Enabled or not StrafeOptimizerState.Active then
        return
    end

    local camera = workspace.CurrentCamera
    if not camera or dt <= 0 then
        return
    end

    local now = os.clock()
    local look = camera.CFrame.LookVector
    local lastLook = StrafeOptimizerState.PreviousLookVector

    if StrafeOptimizerState.AutoTuneRunning and lastLook then
        local dot = math.clamp(lastLook:Dot(look), -1, 1)
        local turnDegrees = math.deg(math.acos(dot))
        local instantRate = turnDegrees / math.max(dt, 1 / 240)

        -- Ignore tiny camera noise and implausible one-frame spikes. Samples are
        -- collected only while the camera is visibly turning.
        if turnDegrees >= 0.04 and instantRate >= 3 and instantRate <= 1200 then
            table.insert(StrafeOptimizerState.TurnRateSamples, instantRate)
            StrafeOptimizerState.AutoTuneSawTurn = true
        end
    end

    StrafeOptimizerState.PreviousLookVector = look

    if not StrafeOptimizerState.AutoTuneRunning then
        return
    end

    if now - StrafeOptimizerState.AutoTuneStartedAt < Settings.StrafeOptimizer.AutoTuneDuration then
        -- Keep countdown/button state fresh without repainting on every frame.
        if now - StrafeOptimizerState.LastStatusUpdateAt >= 0.1 then
            StrafeOptimizerState.LastStatusUpdateAt = now
            UpdateStrafeOptimizerStatus()
        end
        return
    end

    StrafeOptimizerState.AutoTuneRunning = false
    local samples = StrafeOptimizerState.TurnRateSamples

    if #samples >= 8 then
        table.sort(samples)
        local middle = math.floor((#samples + 1) / 2)
        local measuredRate
        if #samples % 2 == 0 then
            measuredRate = (samples[middle] + samples[middle + 1]) / 2
        else
            measuredRate = samples[middle]
        end

        local targetRate = Settings.StrafeOptimizer.TargetTurnRate
        if measuredRate > 0 and targetRate > 0 then
            -- Move toward the target directly, with strict bounds to prevent
            -- an unexpectedly large change from one calibration session.
            local correction = StrafeOptimizerState.AutoTuneStartingMultiplier
                * targetRate / measuredRate
            StrafeOptimizerState.AutoMultiplier = math.clamp(
                correction,
                Settings.StrafeOptimizer.AutoTuneMin,
                Settings.StrafeOptimizer.AutoTuneMax
            )
            StrafeOptimizerState.AutoTuneStatus = "BALANCED"
            ApplyStrafeOptimizerSensitivity()
        else
            StrafeOptimizerState.AutoMultiplier = StrafeOptimizerState.AutoTuneStartingMultiplier
            StrafeOptimizerState.AutoTuneStatus = "NO INPUT"
            ApplyStrafeOptimizerSensitivity()
        end
    else
        -- No meaningful sustained turn was measured; preserve the current setting.
        StrafeOptimizerState.AutoMultiplier = StrafeOptimizerState.AutoTuneStartingMultiplier
        StrafeOptimizerState.AutoTuneStatus = "NO INPUT"
        ApplyStrafeOptimizerSensitivity()
    end

    UpdateStrafeOptimizerStatus()
end)

--========================================================--
-- MOMENTUM PRESERVE LOOP
--========================================================--

Connect(RunService.Heartbeat, function()
    if not Settings.MomentumPreserve.Enabled then
        MomentumPreserveState.WasAirborne = false
        MomentumPreserveState.LastHorizontal =
            Vector3.zero
        return
    end

    if not Root
        or not Root.Parent
        or not Humanoid
        or Humanoid.Health <= 0 then
        return
    end

    local velocity =
        Root.AssemblyLinearVelocity

    local horizontal =
        Vector3.new(
            velocity.X,
            0,
            velocity.Z
        )

    local airborne =
        Humanoid.FloorMaterial ==
        Enum.Material.Air

    if airborne then
        if horizontal.Magnitude >
            MomentumPreserveState.LastHorizontal.Magnitude then

            MomentumPreserveState.LastHorizontal =
                horizontal
        end

        MomentumPreserveState.WasAirborne = true
        return
    end

    if MomentumPreserveState.WasAirborne then
        MomentumPreserveState.WasAirborne = false

        local saved =
            MomentumPreserveState.LastHorizontal

        if saved.Magnitude > 0.05 then
            local strength =
                math.clamp(
                    Settings.MomentumPreserve.Strength,
                    Settings.MomentumPreserve.Min,
                    Settings.MomentumPreserve.Max
                ) / 100

            local currentHorizontal =
                Vector3.new(
                    velocity.X,
                    0,
                    velocity.Z
                )

            local preserved =
                currentHorizontal:Lerp(
                    saved,
                    strength
                )

            Root.AssemblyLinearVelocity =
                Vector3.new(
                    preserved.X,
                    velocity.Y,
                    preserved.Z
                )
        end

        MomentumPreserveState.LastHorizontal =
            Vector3.zero
    end
end)

Connect(RunService.Heartbeat, function()
    if not Settings.BhopAssist.Enabled then
        BhopAssistState.WasAirborne = false
        BhopAssistState.PeakSpeed = 0
        BhopAssistState.PeakDirection = Vector3.zero
        return
    end

    if not Root
        or not Root.Parent
        or not Humanoid
        or Humanoid.Health <= 0 then
        return
    end

    local velocity =
        Root.AssemblyLinearVelocity

    local horizontal =
        Vector3.new(
            velocity.X,
            0,
            velocity.Z
        )

    local airborne =
        Humanoid.FloorMaterial ==
        Enum.Material.Air

    if airborne then
        BhopAssistState.WasAirborne = true

        if horizontal.Magnitude >
            BhopAssistState.PeakSpeed then

            BhopAssistState.PeakSpeed =
                horizontal.Magnitude

            if horizontal.Magnitude > 0.05 then
                BhopAssistState.PeakDirection =
                    horizontal.Unit
            end
        end

        return
    end

    if not BhopAssistState.WasAirborne then
        return
    end

    BhopAssistState.WasAirborne = false

    local now = os.clock()

    if now - BhopAssistState.LastAssist <
        Settings.BhopAssist.Cooldown then

        BhopAssistState.PeakSpeed = 0
        BhopAssistState.PeakDirection =
            Vector3.zero

        return
    end

    local peakSpeed =
        BhopAssistState.PeakSpeed

    if peakSpeed <= 0.05 then
        BhopAssistState.PeakSpeed = 0
        BhopAssistState.PeakDirection =
            Vector3.zero
        return
    end

    local preserve =
        math.clamp(
            Settings.BhopAssist.Preserve,
            Settings.BhopAssist.PreserveMin,
            Settings.BhopAssist.PreserveMax
        ) / 100

    local gain =
        math.clamp(
            Settings.BhopAssist.Gain,
            Settings.BhopAssist.GainMin,
            Settings.BhopAssist.GainMax
        )

    local currentSpeed =
        horizontal.Magnitude

    local restoredSpeed =
        peakSpeed * preserve

    local targetSpeed =
        math.max(
            currentSpeed,
            restoredSpeed
        )

    targetSpeed += gain

    local direction
    local move = Humanoid.MoveDirection

    if move.Magnitude > 0.05 then
        direction =
            Vector3.new(
                move.X,
                0,
                move.Z
            ).Unit
    elseif horizontal.Magnitude > 0.05 then
        direction = horizontal.Unit
    elseif BhopAssistState.PeakDirection.Magnitude > 0.05 then
        direction = BhopAssistState.PeakDirection
    else
        local look = Root.CFrame.LookVector

        direction =
            Vector3.new(
                look.X,
                0,
                look.Z
            )

        if direction.Magnitude > 0.05 then
            direction = direction.Unit
        else
            direction = Vector3.new(0, 0, -1)
        end
    end

    Root.AssemblyLinearVelocity =
        Vector3.new(
            direction.X * targetSpeed,
            velocity.Y,
            direction.Z * targetSpeed
        )

    BhopAssistState.LastAssist = now
    BhopAssistState.PeakSpeed = 0
    BhopAssistState.PeakDirection =
        Vector3.zero
end)

local function FlattenDirection(vector)
    if not vector then
        return nil
    end

    local flat = Vector3.new(vector.X, 0, vector.Z)
    if flat.Magnitude > 0.05 then
        return flat.Unit
    end
    return nil
end

local function ResolveLandingRedirectDirection(currentHorizontal)
    local mode = Settings.LandingRedirect.Direction
    local momentumDirection = FlattenDirection(currentHorizontal)
        or FlattenDirection(LandingRedirectState.PeakDirection)
    local moveDirection = Humanoid and FlattenDirection(Humanoid.MoveDirection)
    local camera = workspace.CurrentCamera
    local cameraDirection = camera and FlattenDirection(camera.CFrame.LookVector)
    local characterDirection = Root and FlattenDirection(Root.CFrame.LookVector)
    local defaultDirection = momentumDirection or moveDirection or cameraDirection or characterDirection
        or Vector3.new(0, 0, -1)

    if mode == "Camera" then
        return cameraDirection or characterDirection or defaultDirection
    elseif mode == "Move" then
        return moveDirection or momentumDirection or cameraDirection or characterDirection or defaultDirection
    end

    -- Momentum mode continues current horizontal travel, falling back to
    -- the fastest direction observed during the airborne portion of the jump.
    return momentumDirection or moveDirection or cameraDirection or characterDirection or defaultDirection
end

Connect(RunService.Heartbeat, function()
    if not Settings.LandingRedirect.Enabled then
        LandingRedirectState.WasAirborne = false
        LandingRedirectState.PeakAirSpeed = 0
        LandingRedirectState.PeakDirection = Vector3.zero
        return
    end

    if not Root or not Root.Parent or not Humanoid or Humanoid.Health <= 0 then
        LandingRedirectState.WasAirborne = false
        LandingRedirectState.PeakAirSpeed = 0
        LandingRedirectState.PeakDirection = Vector3.zero
        return
    end

    local velocity = Root.AssemblyLinearVelocity
    local horizontal = Vector3.new(velocity.X, 0, velocity.Z)
    local airborne = Humanoid.FloorMaterial == Enum.Material.Air

    if airborne then
        LandingRedirectState.WasAirborne = true
        local airSpeed = horizontal.Magnitude
        if airSpeed > LandingRedirectState.PeakAirSpeed then
            LandingRedirectState.PeakAirSpeed = airSpeed
            if airSpeed > 0.05 then
                LandingRedirectState.PeakDirection = horizontal.Unit
            end
        end
        return
    end

    if not LandingRedirectState.WasAirborne then
        return
    end

    LandingRedirectState.WasAirborne = false
    local now = os.clock()
    if LandingRedirectState.LastRedirect > 0
        and now - LandingRedirectState.LastRedirect < Settings.LandingRedirect.Cooldown then
        LandingRedirectState.PeakAirSpeed = 0
        LandingRedirectState.PeakDirection = Vector3.zero
        return
    end

    local currentSpeed = horizontal.Magnitude
    local peakAirSpeed = LandingRedirectState.PeakAirSpeed
    local lostSpeed = math.max(peakAirSpeed - currentSpeed, 0)
    local transfer = math.clamp(
        Settings.LandingRedirect.MomentumTransfer,
        Settings.LandingRedirect.TransferMin,
        Settings.LandingRedirect.TransferMax
    ) / 100
    local power = math.clamp(
        Settings.LandingRedirect.Power,
        Settings.LandingRedirect.PowerMin,
        Settings.LandingRedirect.PowerMax
    )

    -- Recover only the configured share of horizontal speed lost between
    -- the airborne peak and touchdown, then add the selected burst power.
    local targetSpeed = currentSpeed + (lostSpeed * transfer) + power
    if targetSpeed > 0.05 then
        local direction = ResolveLandingRedirectDirection(horizontal)
        if direction and direction.Magnitude > 0.05 then
            Root.AssemblyLinearVelocity = Vector3.new(
                direction.X * targetSpeed,
                velocity.Y,
                direction.Z * targetSpeed
            )
            LandingRedirectState.LastRedirect = now
        end
    end

    LandingRedirectState.PeakAirSpeed = 0
    LandingRedirectState.PeakDirection = Vector3.zero
end)

local fpsElapsed = 0
local fpsFrames = 0
local displayedFPS = 0

Connect(RunService.RenderStepped, function(dt)

    if Settings.VelocityDisplay.Enabled then
        VelocityDisplay.Visible = true

        if Root and Root.Parent then
            local velocity =
                Root.AssemblyLinearVelocity

            local horizontal =
                Vector3.new(
                    velocity.X,
                    0,
                    velocity.Z
                )

            VelocityLabel.Text =
                string.format(
                    "H: %.1f\nV: %.1f\nT: %.1f\nHeight: %.1f",
                    horizontal.Magnitude,
                    velocity.Y,
                    velocity.Magnitude,
                    Root.Position.Y
                )
        else
            VelocityLabel.Text =
                "H: 0.0\nV: 0.0\nT: 0.0\nHeight: 0.0"
        end
    else
        VelocityDisplay.Visible = false
    end

    if Settings.FPSDisplay.Enabled then
        FPSDisplay.Visible = true

        fpsElapsed += dt
        fpsFrames += 1

        if fpsElapsed >= 0.25 then
            displayedFPS =
                math.floor(
                    (fpsFrames / fpsElapsed) + 0.5
                )

            FPSLabel.Text =
                tostring(displayedFPS) .. " FPS"

            fpsElapsed = 0
            fpsFrames = 0
        end
    else
        FPSDisplay.Visible = false
        fpsElapsed = 0
        fpsFrames = 0
    end

    if Settings.Visuals.AngleIndicator then
        local camera = workspace.CurrentCamera
        local look = camera and camera.CFrame.LookVector

        if look then
            local pitch = math.deg(math.asin(math.clamp(look.Y, -1, 1)))
            AngleLabel.Text = string.format("%.1f°", pitch)
        else
            AngleLabel.Text = "0.0°"
        end
    end

    if Settings.Visuals.StrafeReadout then
        if Root and Root.Parent and Humanoid then
            local airborne = Humanoid.FloorMaterial == Enum.Material.Air
            StrafeStatusLabel.Text = airborne and "Airborne" or "Grounded"

            local velocity = Root.AssemblyLinearVelocity
            local horizontalVel = Vector3.new(velocity.X, 0, velocity.Z)

            local move = Humanoid.MoveDirection
            local horizontalMove = Vector3.new(move.X, 0, move.Z)

            if airborne and horizontalVel.Magnitude > 0.5 and horizontalMove.Magnitude > 0.05 then
                local velDir = horizontalVel.Unit
                local moveDir = horizontalMove.Unit

                -- Signed angle of your input relative to current velocity direction.
                local dot = math.clamp(velDir:Dot(moveDir), -1, 1)
                local cross = velDir.X * moveDir.Z - velDir.Z * moveDir.X
                local signedAngle = math.deg(math.acos(dot)) * (cross < 0 and -1 or 1)

                StrafeAngleLabel.Text = string.format("Angle: %.1f°", signedAngle)

                -- Classic strafe-jump sweet spot: input turned ~15-45 degrees
                -- off your current velocity vector. Purely descriptive band,
                -- not a target the script steers you toward.
                local absAngle = math.abs(signedAngle)
                local low, high = 15, 45

                if absAngle < low then
                    StrafeDeltaLabel.Text = string.format("Delta: +%.1f° to band", low - absAngle)
                    StrafeDeltaLabel.TextColor3 = Color3.fromRGB(230, 200, 120)
                elseif absAngle > high then
                    StrafeDeltaLabel.Text = string.format("Delta: -%.1f° over band", absAngle - high)
                    StrafeDeltaLabel.TextColor3 = Color3.fromRGB(230, 140, 120)
                else
                    StrafeDeltaLabel.Text = "Delta: in band"
                    StrafeDeltaLabel.TextColor3 = Color3.fromRGB(140, 220, 150)
                end
            else
                StrafeAngleLabel.Text = "Angle: --"
                StrafeDeltaLabel.Text = "Delta: --"
                StrafeDeltaLabel.TextColor3 = Color3.fromRGB(200, 210, 220)
            end
        end
    end
end)

--========================================================--
-- SPEED CAPPER LOOP (FINAL HORIZONTAL LIMITER)
--========================================================--
-- Apply after physics as well as after the other Heartbeat feature handlers.
-- This prevents wall carry, launch impulses, and Humanoid acceleration from
-- leaving the horizontal velocity above the selected cap for a frame.
do
    local function EnforceSpeedCap()
        if not Settings.SpeedCapper.Enabled then
            return
        end
        if not Root or not Root.Parent or not Humanoid or Humanoid.Health <= 0 then
            return
        end

        local velocity = Root.AssemblyLinearVelocity
        local cappedVelocity = Settings._ClampHorizontalVelocity(velocity)
        if (cappedVelocity - velocity).Magnitude > 0.001 then
            Root.AssemblyLinearVelocity = cappedVelocity
        end
    end

    -- PostSimulation catches physics impulses; Heartbeat is deliberately
    -- registered at the end of the script to catch later feature writes.
    Connect(RunService.PostSimulation, EnforceSpeedCap)
    Connect(RunService.Heartbeat, EnforceSpeedCap)
end
